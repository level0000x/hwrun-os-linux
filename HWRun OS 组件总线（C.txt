HWRun OS 组件总线（Component Bus）完整设计

模块名称： Component Bus (BUS)
版本： 2.0.0
状态： 设计冻结
依赖： 微内核（极简版，~3000行）
依赖协议： METAPROTO, PARAM, LOG, GIT
提供协议： BUS
实现语言： C
设计目标： 插件管理器，是微内核加载的第一个插件，负责发现、安装、加载、启动、停止、卸载、替换所有其他插件

一、设计哲学

1.1 核心定位

组件总线是微内核加载的第一个插件，是所有其他插件的管理者。

```
微内核（~3000行）
    │ 只做三件事：
    │ 1. 启动硬件
    │ 2. 加载第一个插件 → 就是组件总线
    │ 3. 提供最简IPC
    ▼
组件总线（BUS）
    │ 管理所有其他插件：
    │ 1. 发现插件（扫描 plugin.yml）
    │ 2. 安装插件（git clone）
    │ 3. 加载插件（dlopen / insmod）
    │ 4. 启动/停止插件
    │ 5. 卸载插件（可撤销副作用）
    │ 6. 替换插件（运行时热切换）
    │ 7. 协议注册（通过 METAPROTO）
    │ 8. Git 版本控制（所有操作自动 commit）
    ▼
其他插件（FSP, NP, SP, DRIVER, UI, ...）
```

1.2 设计原则

1. 极简依赖：组件总线只依赖微内核提供的最基础功能（任务创建、IPC、内存分配）
2. 自举优先：组件总线自身不依赖任何其他插件，它是一切插件的起点
3. 协议驱动：所有插件间通信通过标准化协议，组件总线本身也注册 BUS 协议
4. 可撤销副作用：所有安装操作可追溯、可逆，卸载时完全清理
5. Git 原生：每个插件是一个 Git 仓库，所有操作自动 commit

五、核心函数实现

5.1 初始化与自举

```c
// bus/src/bus.c

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <dirent.h>
#include <unistd.h>
#include <sys/stat.h>
#include <time.h>

#include "bus.h"
#include "kernel_ipc.h"      // 微内核 IPC 接口
#include "kernel_task.h"     // 微内核任务接口
#include "kernel_mm.h"       // 微内核内存接口

/* 全局组件总线实例 */
static component_bus_t g_bus = {0};

/* ============================================================
   初始化组件总线（被微内核调用）
   ============================================================ */
int bus_init(uint64_t bus_task_id) {
    memset(&g_bus, 0, sizeof(g_bus));
    
    // 1. 记录 BUS 自身的任务 ID
    g_bus.bus_task_id = bus_task_id;
    
    // 2. 设置默认扫描路径
    strcpy(g_bus.scan_paths[0], "/var/lib/hwrun/git/plugins");
    strcpy(g_bus.scan_paths[1], "/usr/lib/hwrun/plugins");
    g_bus.scan_paths_count = 2;
    
    // 3. 初始化 METAPROTO 客户端（通过 IPC）
    // 注意：METAPROTO 已经由微内核加载，BUS 只需要连接它
    g_bus.metaproto_client = metaproto_client_connect();
    if (!g_bus.metaproto_client) {
        // METAPROTO 可能还没就绪，等待
        int retries = 10;
        while (retries-- > 0) {
            yield();
            g_bus.metaproto_client = metaproto_client_connect();
            if (g_bus.metaproto_client) break;
        }
        if (!g_bus.metaproto_client) {
            bus_log("Failed to connect to METAPROTO");
            return -1;
        }
    }
    
    // 4. 注册 BUS 协议（通过 METAPROTO）
    ProtocolInfo proto_info = {
        .name = "BUS",
        .version = "1.0.0",
        .description = "Component Bus Protocol",
        .provider_plugin = "component-bus",
        .status = "active",
    };
    if (metaproto_register(g_bus.metaproto_client, &proto_info) != 0) {
        bus_log("Failed to register BUS protocol");
    }
    
    // 5. 尝试连接其他核心服务
    g_bus.git_storage_client = git_storage_client_connect();
    g_bus.param_client = param_client_connect();
    g_bus.log_client = log_client_connect();
    
    // 6. 设置回调
    g_bus.on_install = bus_default_on_install;
    g_bus.on_uninstall = bus_default_on_uninstall;
    g_bus.on_replace = bus_default_on_replace;
    g_bus.on_state_change = bus_default_on_state_change;
    
    g_bus.initialized = 1;
    bus_log("Component Bus initialized successfully");
    return 0;
}

/* ============================================================
   扫描可用插件
   ============================================================ */
int bus_scan(const char **paths, int path_count) {
    plugin_discovery_t *list = NULL;
    int total = 0;
    
    if (paths == NULL || path_count == 0) {
        paths = (const char **)g_bus.scan_paths;
        path_count = g_bus.scan_paths_count;
    }
    
    bus_log("Scanning plugins...");
    
    for (int i = 0; i < path_count; i++) {
        DIR *dir = opendir(paths[i]);
        if (!dir) continue;
        
        struct dirent *entry;
        while ((entry = readdir(dir)) != NULL) {
            if (strcmp(entry->d_name, ".") == 0 || strcmp(entry->d_name, "..") == 0) {
                continue;
            }
            
            char plugin_path[512];
            snprintf(plugin_path, sizeof(plugin_path), "%s/%s", paths[i], entry->d_name);
            
            struct stat st;
            if (stat(plugin_path, &st) != 0 || !S_ISDIR(st.st_mode)) {
                continue;
            }
            
            char yml_path[512];
            snprintf(yml_path, sizeof(yml_path), "%s/plugin.yml", plugin_path);
            if (access(yml_path, R_OK) != 0) {
                continue;
            }
            
            plugin_discovery_t *discovery = parse_plugin_yml(yml_path);
            if (discovery) {
                discovery->next = list;
                list = discovery;
                total++;
                bus_log("Found plugin: %s (%s)", discovery->id, discovery->version);
            }
        }
        closedir(dir);
    }
    
    bus_log("Scan completed, found %d plugins", total);
    return total;
}
```

5.2 插件安装实现

```c
// bus/src/install.c

/* ============================================================
   安装插件
   ============================================================ */
int bus_install(const char *id, const char *version, int install_deps) {
    if (g_bus.dry_run) {
        bus_log("DRY RUN: Would install plugin %s", id);
        return 0;
    }
    
    // 1. 查找插件
    plugin_discovery_t *discovery = find_discovery(id);
    if (!discovery) {
        bus_log("Plugin %s not found", id);
        return -1;
    }
    
    // 2. 检查是否已安装
    plugin_t *existing = find_plugin(id);
    if (existing && existing->state != PLUGIN_STATE_UNINSTALLED) {
        bus_log("Plugin %s is already installed", id);
        return -2;
    }
    
    // 3. 检查依赖（通过 METAPROTO）
    if (install_deps || g_bus.auto_install_deps) {
        int deps_ok = install_dependencies(discovery);
        if (deps_ok < 0) {
            bus_log("Failed to install dependencies for %s", id);
            return -3;
        }
    }
    
    // 4. 创建插件对象
    plugin_t *p = plugin_create(discovery);
    if (!p) {
        bus_log("Failed to create plugin object for %s", id);
        return -4;
    }
    
    // 5. 克隆 Git 仓库（通过 git-storage 客户端）
    if (discovery->repo[0] != '\0') {
        char repo_path[512];
        snprintf(repo_path, sizeof(repo_path), "/var/lib/hwrun/git/plugins/%s", id);
        
        if (git_storage_clone(g_bus.git_storage_client, discovery->repo, repo_path) != 0) {
            bus_log("Failed to clone repository for %s", id);
            plugin_destroy(p);
            return -5;
        }
        
        if (discovery->tag[0] != '\0') {
            git_storage_checkout(g_bus.git_storage_client, discovery->tag);
        } else if (discovery->branch[0] != '\0') {
            git_storage_checkout(g_bus.git_storage_client, discovery->branch);
        }
    }
    
    // 6. 创建安装记录
    install_record_t *record = calloc(1, sizeof(install_record_t));
    strcpy(record->plugin_id, id);
    record->timestamp = time(NULL);
    record->task_id = g_bus.bus_task_id;
    
    // 7. 复制文件到系统目录
    for (int i = 0; i < discovery->files_count; i++) {
        char src[512], dst[512];
        snprintf(src, sizeof(src), "/var/lib/hwrun/git/plugins/%s/%s", id, discovery->files[i]);
        snprintf(dst, sizeof(dst), "/usr/lib/hwrun/plugins/%s", 
                 strrchr(discovery->files[i], '/') ? strrchr(discovery->files[i], '/') + 1 : discovery->files[i]);
        
        if (copy_file(src, dst) == 0) {
            add_to_record(record, "file", dst);
        }
    }
    
    // 8. 如果是内核模块，加载到内核
    for (int i = 0; i < discovery->files_count; i++) {
        if (strstr(discovery->files[i], ".ko")) {
            char mod_path[512];
            snprintf(mod_path, sizeof(mod_path), "/usr/lib/hwrun/plugins/%s", 
                     strrchr(discovery->files[i], '/') ? strrchr(discovery->files[i], '/') + 1 : discovery->files[i]);
            
            if (load_kernel_module(mod_path) == 0) {
                add_to_record(record, "module", mod_path);
            }
        }
    }
    
    // 9. 如果是 systemd 服务，启动服务
    // （实际实现中需要检查 .service 文件）
    
    // 10. 执行 post_install 钩子
    if (p->post_install[0] != '\0') {
        run_hook(p->post_install, p);
    }
    
    // 11. 更新状态
    p->state = PLUGIN_STATE_INSTALLED;
    p->private_data = record;
    p->task_id = task_create(p->name, NULL);  // 为插件创建微内核任务
    
    // 12. 添加到链表
    p->next = g_bus.plugins;
    g_bus.plugins = p;
    record->next = g_bus.records;
    g_bus.records = record;
    
    // 13. Git commit
    if (g_bus.git_storage_client) {
        char commit_msg[256];
        snprintf(commit_msg, sizeof(commit_msg), "安装插件 %s v%s", id, p->version);
        git_storage_commit(g_bus.git_storage_client, commit_msg);
    }
    
    // 14. 触发回调
    if (g_bus.on_install) {
        g_bus.on_install(p);
    }
    
    bus_log("Plugin %s installed successfully", id);
    return 0;
}
```

5.3 插件加载实现

```c
// bus/src/load.c

/* ============================================================
   加载插件（到内存）
   ============================================================ */
int bus_load(const char *id) {
    plugin_t *p = find_plugin(id);
    if (!p) {
        bus_log("Plugin %s not found", id);
        return -1;
    }
    
    if (p->state != PLUGIN_STATE_INSTALLED) {
        bus_log("Plugin %s is not in installed state", id);
        return -2;
    }
    
    // 1. 检查依赖（通过 METAPROTO）
    for (int i = 0; i < p->requires_count; i++) {
        if (!metaproto_resolve(g_bus.metaproto_client, p->requires[i], NULL)) {
            bus_log("Required protocol %s not available for plugin %s", p->requires[i], id);
            return -3;
        }
    }
    
    // 2. 加载动态库
    void *handle = NULL;
    char lib_path[512];
    
    for (int i = 0; i < p->files_count; i++) {
        if (strstr(p->files[i], ".so")) {
            snprintf(lib_path, sizeof(lib_path), "/usr/lib/hwrun/plugins/%s", 
                     strrchr(p->files[i], '/') ? strrchr(p->files[i], '/') + 1 : p->files[i]);
            
            handle = dlopen(lib_path, RTLD_NOW | RTLD_GLOBAL);
            if (!handle) {
                bus_log("Failed to load plugin %s: %s", id, dlerror());
                return -4;
            }
            
            // 查找插件函数
            char func_name[128];
            snprintf(func_name, sizeof(func_name), "%s_init", id);
            p->init = (int (*)(plugin_t *))dlsym(handle, func_name);
            
            snprintf(func_name, sizeof(func_name), "%s_start", id);
            p->start = (int (*)(plugin_t *))dlsym(handle, func_name);
            
            snprintf(func_name, sizeof(func_name), "%s_stop", id);
            p->stop = (int (*)(plugin_t *))dlsym(handle, func_name);
            
            snprintf(func_name, sizeof(func_name), "%s_destroy", id);
            p->destroy = (int (*)(plugin_t *))dlsym(handle, func_name);
            
            snprintf(func_name, sizeof(func_name), "%s_get_interface", id);
            p->get_interface = (void *(*)(const char *))dlsym(handle, func_name);
            
            p->handle = handle;
            break;
        }
    }
    
    if (!handle) {
        bus_log("No loadable file found for plugin %s", id);
        return -5;
    }
    
    // 3. 调用 init
    if (p->init) {
        if (p->init(p) != 0) {
            bus_log("Plugin %s init failed", id);
            dlclose(handle);
            return -6;
        }
    }
    
    // 4. 注册协议（通过 METAPROTO）
    for (int i = 0; i < p->provides_count; i++) {
        void *impl = p->get_interface ? p->get_interface(p->provides[i]) : NULL;
        if (impl) {
            ProtocolInfo info = {
                .name = p->provides[i],
                .version = "1.0.0",
                .provider_plugin = p->id,
                .status = "active",
            };
            metaproto_register(g_bus.metaproto_client, &info);
            
            protocol_provider_t *provider = calloc(1, sizeof(protocol_provider_t));
            strcpy(provider->protocol, p->provides[i]);
            strcpy(provider->version, "1.0.0");
            strcpy(provider->plugin_id, p->id);
            provider->implementation = impl;
            provider->next = g_bus.providers;
            g_bus.providers = provider;
        }
    }
    
    p->state = PLUGIN_STATE_LOADED;
    p->load_time = time(NULL);
    
    bus_log("Plugin %s loaded successfully", id);
    return 0;
}
```

5.4 插件卸载实现（可撤销副作用）

```c
// bus/src/uninstall.c

/* ============================================================
   卸载插件（可撤销副作用）
   ============================================================ */
int bus_uninstall(const char *id, int force) {
    plugin_t *p = find_plugin(id);
    if (!p) {
        bus_log("Plugin %s not found", id);
        return -1;
    }
    
    // 1. 检查依赖
    if (!force) {
        plugin_t *consumer = find_consumer(id);
        if (consumer) {
            bus_log("Plugin %s is required by %s, use --force", id, consumer->id);
            return -2;
        }
    }
    
    // 2. 如果正在运行，先停止
    if (p->state == PLUGIN_STATE_STARTED) {
        bus_stop(id);
    }
    if (p->state == PLUGIN_STATE_LOADED) {
        bus_unload(id);
    }
    
    // 3. 执行 pre_uninstall
    if (p->pre_uninstall[0] != '\0') {
        run_hook(p->pre_uninstall, p);
    }
    
    // 4. 逆向执行安装记录（可撤销副作用）
    install_record_t *record = find_record(id);
    if (record) {
        // 逆向顺序：最后安装的先卸载
        for (int i = record->services_started_count - 1; i >= 0; i--) {
            stop_service(record->services_started[i]);
            remove_service_file(record->services_started[i]);
        }
        for (int i = record->modules_loaded_count - 1; i >= 0; i--) {
            unload_kernel_module(record->modules_loaded[i]);
        }
        for (int i = record->mount_points_count - 1; i >= 0; i--) {
            umount(record->mount_points[i]);
        }
        for (int i = record->config_changed_count - 1; i >= 0; i--) {
            remove(record->config_changed[i]);
        }
        for (int i = record->files_copied_count - 1; i >= 0; i--) {
            remove(record->files_copied[i]);
        }
    }
    
    // 5. 从 Git 仓库移除
    if (g_bus.git_storage_client) {
        git_storage_remove_submodule(g_bus.git_storage_client, p->id);
    }
    
    // 6. 注销协议
    remove_protocol_providers(p->id);
    
    // 7. 清理微内核任务
    if (p->task_id > 0) {
        task_destroy(p->task_id);
    }
    
    // 8. 从列表移除
    remove_plugin_from_list(p);
    
    // 9. Git commit
    if (g_bus.git_storage_client) {
        char commit_msg[128];
        snprintf(commit_msg, sizeof(commit_msg), "卸载插件 %s", id);
        git_storage_commit(g_bus.git_storage_client, commit_msg);
    }
    
    plugin_destroy(p);
    bus_log("Plugin %s uninstalled successfully", id);
    return 0;
}
```

```c
int bus_service_status(const char *id, plugin_status_t *out) {
    plugin_t *p = find_plugin(id);
    if (!p) return -1;
    out->id = strdup(p->id);
    out->state = p->state;
    out->uptime = (p->state == PLUGIN_STATE_STARTED) ? (time(NULL) - p->load_time) : 0;
    return 0;
}

/* 列出所有插件 */
int bus_service_list(const char *filter, const char *state, 
                     plugin_info_t **out, int *out_count) {
    plugin_t *p = g_bus.plugins;
    int count = 0;
    while (p) {
        if (match_filter(p->id, filter) && match_state(p->state, state)) {
            count++;
        }
        p = p->next;
    }
    *out_count = count;
    return 0;
}

/* 解析协议 */
int bus_service_resolve(const char *protocol, const char *version, 
                        char *out_plugin_id, size_t size) {
    protocol_provider_t *pp = g_bus.providers;
    while (pp) {
        if (strcmp(pp->protocol, protocol) == 0) {
            if (version == NULL || check_version_compatibility(version, pp->version)) {
                strncpy(out_plugin_id, pp->plugin_id, size);
                return 0;
            }
        }
        pp = pp->next;
    }
    // 通过 METAPROTO 查询
    char provider[64];
    if (metaproto_resolve(g_bus.metaproto_client, protocol, version, provider, sizeof(provider)) == 0) {
        strncpy(out_plugin_id, provider, size);
        return 0;
    }
    return -1;
}
```

七、启动流程与自举

7.1 系统启动自举流程

```
微内核 _start()
    │
    ├── 1. 硬件初始化
    ├── 2. 内存分配器初始化
    ├── 3. 中断/时钟初始化
    ├── 4. 加载 METAPROTO 插件（硬编码路径）
    │        └── METAPROTO 注册自己
    ├── 5. plugin_load("/usr/lib/hwrun/plugins/bus.so")
    ├── 6. task_create("bus", entry)
    ├── 7. task_switch(bus)
    │
    ▼
bus_init()  ← BUS 插件启动
    │
    ├── 1. 连接 METAPROTO
    ├── 2. 注册 BUS 协议
    ├── 3. 扫描插件目录
    ├── 4. 加载 PARAM（参数系统）
    ├── 5. 加载 LOG（日志系统）
    ├── 6. 加载 GIT（Git 存储）
    ├── 7. 读取参数，加载默认插件
    │        ├── kernel.type → 内核插件
    │        ├── init.type → init 插件
    │        └── fs.type → 文件系统插件
    └── 8. 系统就绪
```

7.2 BUS 启动时加载的插件顺序

```
BUS 启动后按以下顺序加载插件：

1. PARAM     → 参数系统（必须最先加载，其他插件依赖参数）
2. LOG       → 日志系统（尽早加载，记录后续所有操作）
3. GIT       → Git 存储（尽早加载，记录后续所有变更）
4. 根据参数决定的其他插件：
   ├── kernel.type  → 内核插件（HAP 协议）
   ├── init.type    → init 插件（INIT 协议）
   ├── fs.type      → 文件系统插件（FSP 协议）
   ├── network.type → 网络插件（NP 协议）
   ├── sp.type      → 安全插件（SP 协议）
   ├── scheduler.type → 调度器插件（SCHED 协议）
   └── ui.type      → 用户界面插件（UI 协议）
```

十、总结

10.1 组件总线职责

功能 描述
插件发现 扫描 Git 仓库和本地目录，解析 plugin.yml
插件安装 Git clone 源码，复制文件，加载模块，启动服务
依赖管理 通过 METAPROTO 查询协议提供者，自动安装依赖
生命周期 安装 → 加载 → 启动 → 运行 → 停止 → 卸载
可撤销副作用 记录所有操作，卸载时逆向执行
热替换 运行时替换插件，协议消费者无感知
Git 集成 所有操作自动 commit，可回滚
协议注册 插件加载后自动注册协议到 METAPROTO

10.2 关键设计原则

1. METAPROTO 先启动（自举），BUS 后启动
2. 所有插件操作通过 BUS 协议进行
3. 副作用可撤销（安装记录 + 逆向执行）
4. 热替换通过协议路由切换实现
5. Git 提供版本控制和历史管理
6. BUS 只依赖微内核极简接口（任务、IPC、内存）

10.3 与微内核的接口

```
BUS 从微内核获取：
├── task_create()      → 为每个插件创建微内核任务
├── task_switch()      → 切换任务
├── ipc_send()         → 发送消息到其他插件
├── ipc_recv()         → 接收来自其他插件的消息
├── kmalloc()          → 分配内存
├── kfree()            → 释放内存
└── yield()            → 让出 CPU

BUS 通过微内核 IPC 与其他插件通信：
├── METAPROTO → 协议注册/发现
├── PARAM → 参数读写
├── LOG → 日志写入
└── GIT → Git 操作
```
