HWRun OS INSTANCE 实现文档

模块名称： INSTANCE (Instance Management Protocol)
版本： 1.0.0
状态： 设计冻结
定位： 操作系统实例管理协议，提供容器、MicroVM、完整虚拟机、Unikernel、沙盒的统一管理接口
实现语言： C（内核模块 + 用户态辅助进程）
依赖： 微内核（任务、IPC、内存）、METAPROTO（协议注册）、PARAM（参数读取）、LOG（日志记录）、FSP（文件系统）、PMP（进程管理）、NP（网络协议）、CLUSTER（集群管理）
提供协议： INSTANCE

一、设计哲学

1.1 核心定位

INSTANCE 是 HWRun OS 的“操作系统实例工厂”——所有与操作系统实例（容器、MicroVM、虚拟机、Unikernel、沙盒）相关的操作都通过 INSTANCE 协议进行。

```
INSTANCE = 操作系统实例管理协议 = 所有实例操作的统一入口

上层插件 (SANDBOX, CONTAINER, CLUSTER, PERMISSION, MONITOR, ...)
    │
    │  通过 INSTANCE 协议管理实例
    ▼
INSTANCE 插件
    │
    │  统一实例管理层
    │  ├── 实例生命周期 (创建/销毁/启动/停止/暂停/恢复)
    │  ├── 实例资源管理 (CPU/内存/GPU/存储/网络)
    │  ├── 实例快照 (创建/恢复/删除)
    │  ├── 实例迁移 (跨节点)
    │  └── 实例监控 (状态/资源使用)
    │
    ├── 容器后端 (runc)        → 进程级隔离，共享内核[reference:0]
    ├── 系统容器后端 (LXD)     → 操作系统级隔离，共享内核
    ├── MicroVM 后端 (Firecracker) → 硬件级隔离，独立内核，<125ms 启动[reference:1]
    ├── 完整 VM 后端 (KVM)     → 完整硬件虚拟化，最强隔离[reference:2]
    ├── Unikernel 后端         → 应用直接跑在内核态，极致轻量[reference:3]
    └── 沙盒后端 (gVisor)      → 用户态内核拦截，中等隔离[reference:4]
    │
    ▼
硬件层 / 内核层
```

1.2 设计原则

1. 协议驱动：所有实例操作通过 INSTANCE 协议进行
2. 后端可插拔：容器、MicroVM、虚拟机、Unikernel、沙盒都是可替换的后端
3. 统一抽象：所有实例类型通过同一套接口管理
4. 用户选择：用户通过配置选择实例类型，无需改变操作方式
5. 资源隔离：每个实例有独立的 CPU/内存/GPU/存储/网络资源
6. 生命周期管理：完整的创建→启动→运行→停止→销毁生命周期

1.3 实例类型对比

特性 容器 (runc) 系统容器 (LXD) MicroVM (Firecracker) 完整 VM (KVM) Unikernel 沙盒 (gVisor)
内核 共享宿主机 共享宿主机 独立内核 独立内核 独立内核 用户态内核
隔离级别 进程级 操作系统级 硬件级 硬件级 硬件级 系统调用级
启动时间 ~100ms ~500ms <125ms ~10s+ ~50ms ~200ms
内存开销 ~10-50MB ~50-200MB ~5MB ~500MB+ ~2-10MB ~50-100MB
安全边界 较弱 中等 强 最强 强 中等
设备支持 基础 完整 基础 (virtio) 完整 有限 基础
适用场景 微服务 多租户OS Serverless 完整环境 极简应用 安全容器

2.2 INSTANCE 在 HWRun OS 中的位置

```
系统启动顺序：
BIOS/UEFI → 引导加载器 → 微内核 → METAPROTO → BUS → PARAM → LOG → GIT → HAP → PMP → FSP → NP → SP → LOADER → CLUSTER → INSTANCE → 其他插件
                                                                                                              ↑
                                                                                                   INSTANCE 在这里加载
```

INSTANCE 是第八个加载的核心功能插件（HAP → PMP → FSP → NP → SP → LOADER → CLUSTER → INSTANCE）。在它之后，依赖实例管理的插件（SANDBOX、CONTAINER 等）才能正常工作。

2.3 INSTANCE 的上层消费者

上层插件 需要 INSTANCE 提供什么
SANDBOX（沙盒） 轻量级沙盒环境
CONTAINER（容器） 容器运行时隔离
CLUSTER（集群） 跨节点实例调度
PERMISSION（权限） 实例级权限控制
MONITOR（监控） 实例状态监控

5.1 模块初始化和加载

```c
// instance/src/instance.c

#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/init.h>
#include <linux/slab.h>
#include <linux/string.h>

#include "instance.h"
#include "instance_ipc.h"
#include "instance_manager.h"
#include "instance_backend_runc.h"
#include "instance_backend_lxc.h"
#include "instance_backend_firecracker.h"
#include "instance_backend_kvm.h"
#include "instance_backend_unikernel.h"
#include "instance_backend_gvisor.h"

MODULE_LICENSE("GPL");
MODULE_AUTHOR("HWRun OS");
MODULE_DESCRIPTION("HWRun OS Instance Management Protocol Plugin");
MODULE_VERSION("1.0.0");

static instance_context_t g_instance_ctx = {0};

/* ============================================================
   模块加载入口
   ============================================================ */
static int __init instance_init(void) {
    int ret;
    
    printk(KERN_INFO "HWRun INSTANCE: Initializing...\n");
    
    // 1. 初始化上下文
    memset(&g_instance_ctx, 0, sizeof(instance_context_t));
    
    // 2. 注册后端
    instance_backend_register(&runc_backend);
    instance_backend_register(&lxc_backend);
    instance_backend_register(&firecracker_backend);
    instance_backend_register(&kvm_backend);
    instance_backend_register(&unikernel_backend);
    instance_backend_register(&gvisor_backend);
    
    // 3. 注册 IPC 消息处理器
    ret = instance_ipc_register();
    if (ret < 0) {
        printk(KERN_ERR "HWRun INSTANCE: Failed to register IPC handler\n");
        return ret;
    }
    
    // 4. 注册 INSTANCE 协议（通过 METAPROTO - 通过 IPC）
    instance_register_protocol();
    
    g_instance_ctx.initialized = 1;
    printk(KERN_INFO "HWRun INSTANCE: Initialized successfully\n");
    printk(KERN_INFO "  Supported backends: Container, System Container, MicroVM, VM, Unikernel, Sandbox\n");
    
    return 0;
}

/* ============================================================
   模块卸载入口
   ============================================================ */
static void __exit instance_exit(void) {
    printk(KERN_INFO "HWRun INSTANCE: Exiting...\n");
    
    instance_ipc_unregister();
    instance_cleanup_instances();
    
    g_instance_ctx.initialized = 0;
    printk(KERN_INFO "HWRun INSTANCE: Exited\n");
}

module_init(instance_init);
module_exit(instance_exit);
```

5.2 实例创建

```c
// instance/src/instance_manager.c

#include <linux/kernel.h>
#include <linux/slab.h>
#include <linux/string.h>
#include <linux/random.h>
#include <linux/time.h>

#include "instance.h"
#include "instance_ipc.h"
#include "instance_backend.h"

/* ============================================================
   生成实例 ID
   ============================================================ */
static void instance_generate_id(char *out_id, size_t size) {
    uint32_t rand1, rand2;
    get_random_bytes(&rand1, sizeof(rand1));
    get_random_bytes(&rand2, sizeof(rand2));
    snprintf(out_id, size, "inst-%08x%08x", rand1, rand2);
}

/* ============================================================
   创建实例
   ============================================================ */
int instance_create(instance_config_t *config, char *out_id, size_t id_size) {
    instance_backend_t *backend;
    int ret;
    
    // 1. 根据类型选择后端
    backend = instance_backend_find(config->type);
    if (!backend) {
        return -ENOENT;
    }
    
    // 2. 生成实例 ID
    char instance_id[64];
    instance_generate_id(instance_id, sizeof(instance_id));
    strncpy(out_id, instance_id, id_size - 1);
    
    // 3. 调用后端创建
    ret = backend->create(config, instance_id);
    if (ret < 0) {
        return ret;
    }
    
    // 4. 创建实例信息
    instance_info_t *info = kmalloc(sizeof(instance_info_t), GFP_KERNEL);
    if (!info) {
        backend->destroy(instance_id);
        return -ENOMEM;
    }
    
    memset(info, 0, sizeof(instance_info_t));
    strncpy(info->id, instance_id, sizeof(info->id) - 1);
    strncpy(info->name, config->name, sizeof(info->name) - 1);
    info->type = config->type;
    info->state = INSTANCE_STATE_CREATED;
    info->created_at = get_jiffies_64();
    
    // 5. 添加到实例列表
    info->next = g_instance_ctx.instances;
    g_instance_ctx.instances = info;
    g_instance_ctx.instance_count++;
    
    // 6. 如果配置了自动启动，启动实例
    if (config->auto_start) {
        instance_start(instance_id);
    }
    
    printk(KERN_INFO "HWRun INSTANCE: Created %s (type=%d, id=%s)\n",
           config->name, config->type, instance_id);
    
    return 0;
}

/* ============================================================
   启动实例
   ============================================================ */
int instance_start(const char *instance_id) {
    instance_info_t *info = instance_find(instance_id);
    if (!info) {
        return -ENOENT;
    }
    
    instance_backend_t *backend = instance_backend_find(info->type);
    if (!backend) {
        return -ENOENT;
    }
    
    info->state = INSTANCE_STATE_STARTING;
    int ret = backend->start(instance_id);
    if (ret < 0) {
        info->state = INSTANCE_STATE_ERROR;
        return ret;
    }
    
    info->state = INSTANCE_STATE_RUNNING;
    info->started_at = get_jiffies_64();
    
    printk(KERN_INFO "HWRun INSTANCE: Started %s\n", instance_id);
    return 0;
}

/* ============================================================
   停止实例
   ============================================================ */
int instance_stop(const char *instance_id, int force) {
    instance_info_t *info = instance_find(instance_id);
    if (!info) {
        return -ENOENT;
    }
    
    instance_backend_t *backend = instance_backend_find(info->type);
    if (!backend) {
        return -ENOENT;
    }
    
    info->state = INSTANCE_STATE_STOPPING;
    int ret = backend->stop(instance_id);
    if (ret < 0) {
        info->state = INSTANCE_STATE_ERROR;
        return ret;
    }
    
    info->state = INSTANCE_STATE_STOPPED;
```

5.8 IPC 消息处理

```c
// instance/src/instance_ipc.c

#include <linux/slab.h>
#include <linux/uaccess.h>

#include "instance.h"
#include "instance_ipc.h"
#include "instance_manager.h"

/* ============================================================
   IPC 消息处理入口
   ============================================================ */
void instance_ipc_handler(uint64_t from_task, void *data, size_t len) {
    uint32_t msg_type = *(uint32_t *)data;
    
    switch (msg_type) {
        case INSTANCE_MSG_CREATE: {
            instance_create_req_t *req = (instance_create_req_t *)data;
            instance_create_resp_t resp;
            int ret = instance_create(&req->config, resp.instance_id,
                                       sizeof(resp.instance_id));
            resp.error = ret;
            ipc_reply(from_task, &resp, sizeof(resp));
            break;
        }
        
        case INSTANCE_MSG_DESTROY: {
            char *instance_id = (char *)data;
            int ret = instance_destroy(instance_id, 0);
            ipc_reply(from_task, &ret, sizeof(ret));
            break;
        }
        
        case INSTANCE_MSG_START: {
            char *instance_id = (char *)data;
            int ret = instance_start(instance_id);
            ipc_reply(from_task, &ret, sizeof(ret));
            break;
        }
        
        case INSTANCE_MSG_STOP: {
            char *instance_id = (char *)data;
            int ret = instance_stop(instance_id, 0);
            ipc_reply(from_task, &ret, sizeof(ret));
            break;
        }
        
        case INSTANCE_MSG_LIST: {
            instance_info_t *list;
            uint32_t count;
            instance_list(&list, &count);
            ipc_reply(from_task, list, count * sizeof(instance_info_t));
            break;
        }
        
        case INSTANCE_MSG_INFO: {
            char *instance_id = (char *)data;
            instance_info_t info;
            int ret = instance_info(instance_id, &info);
            if (ret == 0) {
                ipc_reply(from_task, &info, sizeof(info));
            } else {
                ipc_reply(from_task, NULL, 0);
            }
            break;
        }
        
        default:
            printk(KERN_WARNING "HWRun INSTANCE: Unknown IPC message type: %u\n",
                   msg_type);
            break;
    }
}
```

七、总结

维度 决策
定位 操作系统实例管理协议，所有实例操作的统一入口
实现方式 Linux 内核模块 + 用户态后端
代码量 ~1500-2000 行 C
提供协议 INSTANCE (v1.0)
依赖 微内核 IPC + METAPROTO + PARAM + LOG + FSP + PMP + NP + CLUSTER
加载时机 在 SANDBOX、CONTAINER 等依赖实例管理的插件之前
支持后端 容器 (runc)、系统容器 (LXD)、MicroVM (Firecracker)、完整 VM (KVM)、Unikernel、沙盒 (gVisor)

INSTANCE 是 HWRun OS 中第八个加载的核心功能插件（HAP → PMP → FSP → NP → SP → LOADER → CLUSTER → INSTANCE）。它通过统一的 INSTANCE 协议向上层插件提供容器、MicroVM、完整虚拟机、Unikernel、沙盒等多种操作系统实例的统一管理接口。用户通过配置选择实例类型，操作方式完全一致。INSTANCE 协议使 HWRun OS 实现了从进程级隔离到硬件级隔离的完整实例管理能力。
