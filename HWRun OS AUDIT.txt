HWRun OS AUDIT 实现文档

模块名称： AUDIT (Audit Protocol)
版本： 1.0.0
状态： 设计冻结
定位： 审计协议，提供系统安全审计、操作日志、合规追踪、事件监控等完整的审计能力
实现语言： C（内核模块 + 用户态服务）
依赖： 微内核（任务、IPC、内存）、METAPROTO（协议注册）、PARAM（参数读取）、LOG（日志记录）、FSP（文件系统协议）、SP（安全协议）、PERMISSION（权限管理）
提供协议： AUDIT

一、设计哲学

1.1 核心定位

AUDIT 是 HWRun OS 的审计核心——记录系统中所有安全相关事件，提供完整的审计追踪和合规性保障。

```
AUDIT = 审计协议 = 所有安全审计操作的统一入口

安全事件 (权限变更、登录、配置修改、文件访问、进程创建...)
    │
    │  通过 AUDIT 协议记录
    ▼
AUDIT 插件
    │
    │  基于 Linux 内核 audit 子系统
    │  ├── 事件收集: 内核审计事件
    │  ├── 事件过滤: 规则匹配
    │  ├── 事件存储: 持久化存储
    │  └── 事件查询: 检索和分析
    │
    ▼
持久化存储 (审计日志文件 / STORAGE 卷)
```

1.2 设计原则

1. 全面覆盖：记录所有安全相关事件（登录、权限变更、配置修改、文件访问、进程创建、网络连接等）
2. 不可篡改：审计日志以只追加方式写入，防止篡改
3. 规则可配：支持灵活的审计规则配置（基于用户、操作、资源等）
4. 高效低噪：通过规则过滤减少无关事件，保证系统性能
5. 合规就绪：支持常见合规标准（PCI-DSS、HIPAA、GDPR、SOX 等）
6. 可查询：支持按时间、用户、事件类型等维度检索审计日志

1.3 审计事件分类

类别 事件类型 说明
认证事件 AUTH_SUCCESS, AUTH_FAILURE 用户登录成功/失败
授权事件 PERMISSION_GRANT, PERMISSION_REVOKE 权限授予/撤销
用户管理 USER_ADD, USER_REMOVE, USER_MODIFY 用户创建/删除/修改
配置变更 CONFIG_CHANGE 系统配置修改
文件访问 FILE_READ, FILE_WRITE, FILE_DELETE 文件读写删除
进程管理 PROCESS_START, PROCESS_STOP 进程创建/终止
网络连接 NETWORK_CONNECT, NETWORK_DISCONNECT 网络连接
系统操作 SYSTEM_REBOOT, SYSTEM_SHUTDOWN 系统重启/关机
插件管理 PLUGIN_INSTALL, PLUGIN_REMOVE 插件安装/卸载
实例管理 INSTANCE_CREATE, INSTANCE_DESTROY 实例创建/销毁
安全管理 SECURITY_ALERT 安全告警

2.2 AUDIT 在 HWRun OS 中的位置

```
系统启动顺序：
BIOS/UEFI → 引导加载器 → 微内核 → METAPROTO → BUS → PARAM → LOG → GIT → HAP → PMP → FSP → NP → SP → LOADER → CLUSTER → INSTANCE → SANDBOX → INPUT → DISPLAY → PERMISSION → STORAGE → NODE_DISCOVERY → CONSENSUS → FS_TRANSFER → AUDIT → 其他插件
                                                                                                                                            ↑
                                                                                                                                  AUDIT 在这里加载
```

AUDIT 在 FS_TRANSFER 之后、上层应用插件之前加载，为系统提供审计能力。

2.3 AUDIT 的上层消费者

上层插件 需要 AUDIT 提供什么
PERMISSION (权限管理) 权限变更审计
SP (安全协议) 安全事件审计
PKGMGR (包管理) 包安装/卸载审计
CLUSTER (集群管理) 集群操作审计
MONITOR (监控) 安全事件监控

```c
// audit/src/audit.c

#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/init.h>
#include <linux/slab.h>
#include <linux/string.h>
#include <linux/fs.h>
#include <linux/audit.h>
#include <linux/security.h>

#include "audit.h"
#include "audit_ipc.h"
#include "audit_events.h"
#include "audit_rules.h"
#include "audit_storage.h"
#include "audit_hooks.h"

MODULE_LICENSE("GPL");
MODULE_AUTHOR("HWRun OS");
MODULE_DESCRIPTION("HWRun OS Audit Protocol Plugin");
MODULE_VERSION("1.0.0");

static audit_context_t g_audit_ctx = {0};

/* ============================================================
   模块加载入口
   ============================================================ */
static int __init audit_init(void) {
    int ret;
    
    printk(KERN_INFO "HWRun AUDIT: Initializing...\n");
    
    // 1. 初始化上下文
    memset(&g_audit_ctx, 0, sizeof(audit_context_t));
    
    // 2. 设置审计日志路径
    strcpy(g_audit_ctx.log_path, "/var/log/hwrun/audit.log");
    g_audit_ctx.max_log_size = 100 * 1024 * 1024;  // 100MB
    g_audit_ctx.max_log_files = 10;
    
    // 3. 初始化审计规则
    audit_init_default_rules(&g_audit_ctx);
    
    // 4. 注册 LSM 钩子
    ret = audit_hooks_register();
    if (ret < 0) {
        printk(KERN_WARNING "HWRun AUDIT: Failed to register LSM hooks\n");
        // 继续运行，内核审计功能可能不可用
    }
    
    // 5. 注册 IPC 消息处理器
    ret = audit_ipc_register();
    if (ret < 0) {
        printk(KERN_ERR "HWRun AUDIT: Failed to register IPC handler\n");
        return ret;
    }
    
    // 6. 注册 AUDIT 协议
    audit_register_protocol();
    
    g_audit_ctx.initialized = 1;
    printk(KERN_INFO "HWRun AUDIT: Initialized successfully\n");
    printk(KERN_INFO "  Audit log: %s\n", g_audit_ctx.log_path);
    printk(KERN_INFO "  Max log size: %lu MB, max files: %d\n",
           g_audit_ctx.max_log_size / (1024 * 1024),
           g_audit_ctx.max_log_files);
    
    return 0;
}

/* ============================================================
   模块卸载入口
   ============================================================ */
static void __exit audit_exit(void) {
    printk(KERN_INFO "HWRun AUDIT: Exiting...\n");
    
    audit_ipc_unregister();
    audit_hooks_unregister();
    
    g_audit_ctx.initialized = 0;
    printk(KERN_INFO "HWRun AUDIT: Exited\n");
}

module_init(audit_init);
module_exit(audit_exit);
```

5.2 审计事件记录

```c
// audit/src/audit_events.c

#include <linux/kernel.h>
#include <linux/slab.h>
#include <linux/string.h>
#include <linux/time.h>
#include <linux/cred.h>
#include <linux/sched.h>
#include <linux/uidgid.h>

#include "audit.h"
#include "audit_events.h"
#include "audit_storage.h"
#include "audit_rules.h"

/* ============================================================
   记录审计事件
   ============================================================ */
int audit_log_event(audit_event_type_t type, audit_severity_t severity,
                     const char *target, const char *action,
                     int result, const char *reason,
                     const uint8_t *data, uint32_t data_len) {
    struct audit_event *event;
    struct task_struct *task;
    const struct cred *cred;
    int ret;
    
    // 1. 检查是否需要记录 (规则过滤)
    if (!audit_should_log(type, severity, target)) {
        return 0;  // 被规则过滤，不记录
    }
    
    // 2. 检查速率限制
    if (audit_rate_limit_exceeded()) {
        return -EBUSY;
    }
    
    // 3. 创建审计事件
    event = kmalloc(sizeof(struct audit_event), GFP_KERNEL);
    if (!event) {
        return -ENOMEM;
    }
    
    memset(event, 0, sizeof(struct audit_event));
    
    // 4. 填充事件信息
    event->id = audit_generate_id();
    event->type = type;
    event->severity = severity;
    event->timestamp = audit_get_timestamp_ns();
    
    // 获取当前任务信息
    task = current;
    cred = get_task_cred(task);
    if (cred) {
        event->uid = from_kuid(&init_user_ns, cred->uid);
        event->gid = from_kgid(&init_user_ns, cred->gid);
        // 获取用户名 (简化实现)
        snprintf(event->user, sizeof(event->user), "uid-%d", event->uid);
        put_cred(cred);
    }
    
    event->task_id = (uint64_t)task->pid;
    
    if (target) {
        strncpy(event->target, target, sizeof(event->target) - 1);
    }
    if (action) {
        strncpy(event->action, action, sizeof(event->action) - 1);
    }
    event->result = result;
    if (reason) {
        strncpy(event->reason, reason, sizeof(event->reason) - 1);
    }
    
    if (data && data_len > 0) {
        event->data = kmalloc(data_len, GFP_KERNEL);
        if (event->data) {
            memcpy(event->data, data, data_len);
            event->data_len = data_len;
        }
    }
    
    // 5. 持久化存储
    ret = audit_storage_write(event);
    if (ret < 0) {
        kfree(event->data);
        kfree(event);
        return ret;
    }
    
    // 6. 添加到缓存 (用于查询)
    event->next = g_audit_ctx.events;
    g_audit_ctx.events = event;
    g_audit_ctx.event_count++;
    
    // 7. 触发告警 (严重事件)
    if (severity >= AUDIT_SEVERITY_CRITICAL) {
        audit_trigger_alert(event);
    }
    
    return 0;
}

/* ============================================================
   检查是否需要记录 (规则过滤)
   ============================================================ */
int audit_should_log(audit_event_type_t type, audit_severity_t severity,
                      const char *target) {
    audit_rule_t *rule = g_audit_ctx.rules;
    
    while (rule) {
        if (!rule->enabled) {
            rule = rule->next;
            continue;
        }
        
        // 检查严重级别
        if (severity < rule->min_severity) {
            rule = rule->next;
            continue;
        }
        
        // 检查事件类型
        int type_match = 0;
        for (int i = 0; i < rule->type_count; i++) {
            if (rule->types[i] == type) {
                type_match = 1;
                break;
            }
        }
        if (!type_match) {
            rule = rule->next;
            continue;
        }
        
        // 检查目标路径 (通配符匹配)
        if (rule->path_count > 0 && target) {
            int path_match = 0;
            for (int i = 0; i < rule->path_count; i++) {
                if (audit_path_match(target, rule->paths[i])) {
                    path_match = 1;
                    break;
                }
            }
            if (!path_match) {
                rule = rule->next;
                continue;
            }
        }
        
        // 所有条件满足，需要记录
        return 1;
    }
    
    // 没有匹配的规则，默认不记录
    return 0;
}
```

```c
// audit/src/audit_storage.c

#include <linux/kernel.h>
#include <linux/slab.h>
#include <linux/string.h>
#include <linux/fs.h>
#include <linux/file.h>
#include <linux/uaccess.h>
#include <linux/seq_file.h>

#include "audit.h"
#include "audit_storage.h"

/* ============================================================
   写入审计事件到日志文件
   ============================================================ */
int audit_storage_write(struct audit_event *event) {
    struct file *filp;
    char buf[1024];
    ssize_t written;
    loff_t pos;
    int ret;
    
    // 1. 检查日志文件大小，触发轮转
    if (g_audit_ctx.log_size > g_audit_ctx.max_log_size) {
        audit_storage_rotate();
    }
    
    // 2. 打开审计日志文件
    filp = filp_open(g_audit_ctx.log_path,
                     O_WRONLY | O_CREAT | O_APPEND, 0644);
    if (IS_ERR(filp)) {
        return PTR_ERR(filp);
    }
    
    // 3. 格式化审计事件
    audit_format_event(event, buf, sizeof(buf));
    
    // 4. 写入文件
    pos = filp->f_pos;
    written = kernel_write(filp, buf, strlen(buf), &pos);
    if (written < 0) {
        filp_close(filp, NULL);
        return written;
    }
    
    g_audit_ctx.log_size += written;
    
    filp_close(filp, NULL);
    return 0;
}

/* ============================================================
   格式化审计事件
   ============================================================ */
void audit_format_event(struct audit_event *event, char *buf, size_t size) {
    char time_str[32];
    struct tm tm;
    time_t t = event->timestamp / 1000000000ULL;
    struct timespec ts = {.tv_sec = t, .tv_nsec = event->timestamp % 1000000000ULL};
    
    time_to_tm(t, 0, &tm);
    strftime(time_str, sizeof(time_str), "%Y-%m-%d %H:%M:%S", &tm);
    
    char *type_str = audit_get_type_string(event->type);
    char *sev_str = audit_get_severity_string(event->severity);
    
    snprintf(buf, size,
             "[%s.%09ld] [%s] [%s] uid=%d user=%s target=%s action=%s result=%d%s%s\n",
             time_str, ts.tv_nsec,
             sev_str,
             type_str,
             event->uid,
             event->user,
             event->target,
             event->action,
             event->result,
             event->reason[0] ? " reason=" : "",
             event->reason);
}
```

```c
void audit_init_default_rules(audit_context_t *ctx) {

    audit_rule_t *rule;
    
    // 规则 1: 记录所有认证事件

    if (rule) {
        memset(rule, 0, sizeof(audit_rule_t));
        strcpy(rule->id, "auth-events");
        strcpy(rule->name, "Authentication Events");
        rule->enabled = 1;
        rule->min_severity = AUDIT_SEVERITY_INFO;
        rule->types = kmalloc(2 * sizeof(audit_event_type_t), GFP_KERNEL);
        if (rule->types) {
            rule->types[0] = AUDIT_EVENT_AUTH_SUCCESS;
            rule->types[1] = AUDIT_EVENT_AUTH_FAILURE;
            rule->type_count = 2;
        }
        rule->next = ctx->rules;
        ctx->rules = rule;
    }
    
    // 规则 2: 记录权限变更
    rule = kmalloc(sizeof(audit_rule_t), GFP_KERNEL);
    if (rule) {
        memset(rule, 0, sizeof(audit_rule_t));
        strcpy(rule->id, "perm-events");
        strcpy(rule->name, "Permission Changes");
        rule->enabled = 1;
        rule->min_severity = AUDIT_SEVERITY_WARN;
        rule->types = kmalloc(2 * sizeof(audit_event_type_t), GFP_KERNEL);
        if (rule->types) {
            rule->types[0] = AUDIT_EVENT_PERMISSION_GRANT;
            rule->types[1] = AUDIT_EVENT_PERMISSION_REVOKE;
            rule->type_count = 2;
        }
        rule->next = ctx->rules;
        ctx->rules = rule;
    }
    
    // 规则 3: 记录系统配置变更
    rule = kmalloc(sizeof(audit_rule_t), GFP_KERNEL);
    if (rule) {
        memset(rule, 0, sizeof(audit_rule_t));
        strcpy(rule->id, "config-events");
        strcpy(rule->name, "Configuration Changes");
        rule->enabled = 1;
        rule->min_severity = AUDIT_SEVERITY_WARN;
        rule->types = kmalloc(sizeof(audit_event_type_t), GFP_KERNEL);
        if (rule->types) {
            rule->types[0] = AUDIT_EVENT_CONFIG_CHANGE;
            rule->type_count = 1;
        }
        rule->next = ctx->rules;
        ctx->rules = rule;
    }
}

/* ============================================================
   添加审计规则
   ============================================================ */
int audit_add_rule(audit_rule_t *new_rule, char *out_id) {
    audit_rule_t *rule;
    
    // 复制规则
    rule = kmalloc(sizeof(audit_rule_t), GFP_KERNEL);
    if (!rule) {
        return -ENOMEM;
    }
    
    memcpy(rule, new_rule, sizeof(audit_rule_t));
    rule->next = g_audit_ctx.rules;
    g_audit_ctx.rules = rule;
    
    if (out_id) {
        strncpy(out_id, rule->id, 63);
    }
    
    return 0;
}
```

5.6 IPC 消息处理

```c
// audit/src/audit_ipc.c

#include <linux/slab.h>
#include <linux/uaccess.h>

#include "audit.h"
#include "audit_ipc.h"
#include "audit_events.h"
#include "audit_rules.h"
#include "audit_storage.h"

/* ============================================================
   IPC 消息处理入口
   ============================================================ */
void audit_ipc_handler(uint64_t from_task, void *data, size_t len) {
    uint32_t msg_type = *(uint32_t *)data;
    
    switch (msg_type) {
        case AUDIT_MSG_LOG: {
            audit_log_req_t *req = (audit_log_req_t *)data;
            audit_log_resp_t resp;
            int ret = audit_log_event(req->type, req->severity,
                                       req->target, req->action,
                                       req->result, req->reason,
                                       req->data, req->data_len);
            resp.success = (ret == 0);
            resp.event_id = (ret == 0) ? audit_generate_id() : 0;
            resp.error = ret;
            ipc_reply(from_task, &resp, sizeof(resp));
            break;
        }
        
        case AUDIT_MSG_QUERY: {
            audit_query_req_t *req = (audit_query_req_t *)data;
            audit_query_resp_t resp;
            // 查询审计日志
            // 从存储中读取符合条件的日志条目
            ipc_reply(from_task, &resp, sizeof(resp));
            break;
        }
        
        case AUDIT_MSG_ADD_RULE: {
            audit_rule_t *rule = (audit_rule_t *)data;
            char rule_id[64];
            int ret = audit_add_rule(rule, rule_id);
            ipc_reply(from_task, &ret, sizeof(ret));
            break;
        }
        
        case AUDIT_MSG_GET_POLICY: {
            // 返回所有规则
            ipc_reply(from_task, g_audit_ctx.rules, 0);
            break;
        }
        
        case AUDIT_MSG_ROTATE: {
            int ret = audit_storage_rotate();
            ipc_reply(from_task, &ret, sizeof(ret));
            break;
        }
        
        default:
            printk(KERN_WARNING "HWRun AUDIT: Unknown IPC message type: %u\n",
                   msg_type);
            break;
    }
}
```

十、总结

维度 决策
定位 审计协议，所有安全审计的统一入口
实现方式 Linux 内核模块，基于 LSM 钩子
代码量 ~1200-1500 行 C
提供协议 AUDIT (v1.0)
依赖 微内核 IPC + METAPROTO + PARAM + LOG + FSP + SP + PERMISSION
审计事件 认证、授权、用户管理、配置变更、文件访问、进程管理、网络连接、插件管理、实例管理等
合规支持 PCI-DSS、HIPAA、GDPR、SOX、ISO 27001

AUDIT 是 HWRun OS 中审计的核心插件。它基于 Linux 内核的 LSM 钩子，通过 AUDIT 协议向上层插件提供完整的安全审计能力。AUDIT 记录系统中所有安全相关事件，通过灵活的审计规则控制记录范围，通过持久化存储保证审计日志的完整性和不可篡改性。上层插件（PERMISSION、SP、PKGMGR、CLUSTER、MONITOR 等）通过 AUDIT 协议记录审计事件，不直接操作内核审计子系统。
