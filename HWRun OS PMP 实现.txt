HWRun OS PMP 实现文档

模块名称： PMP (Process Management Protocol)
版本： 1.0.0
状态： 设计冻结
定位： 进程管理协议，提供对进程、线程、调度、信号等操作系统核心功能的管理
实现语言： C（内核模块）
依赖： 微内核（任务、IPC、内存）、METAPROTO（协议注册）、PARAM（参数读取）、LOG（日志记录）、HAP（硬件抽象）
提供协议： PMP

一、设计哲学

1.1 核心定位

PMP 是 HWRun OS 的进程管理核心——所有与进程、线程、调度、信号相关的操作都通过 PMP 协议进行。

```
PMP = 进程管理协议 = 所有进程相关操作 (HAP) 的统一入口

上层插件 (SCHED, SHELL, CLUSTER, SANDBOX, ...)
    │
    │  通过 PMP 协议管理进程
    ▼
PMP 插件 (内核模块)
    │
    │  直接操作内核 task_struct
    ▼
Linux 内核 (进程调度器)
```

1.2 设计原则

1. 协议驱动：所有进程操作通过 PMP 协议进行
2. 内核实现：PMP 插件运行在内核态，直接操作 task_struct
3. 进程/线程统一：Linux 内核中进程和线程都是 task_struct
4. 标准接口：暴露 fork/clone/exec/wait/kill/signal 等标准 POSIX 接口
5. 调度抽象：提供调度策略和优先级设置接口

1.3 PMP 插件的本质

PMP 插件 = Linux 内核进程管理接口的 HWRun 协议封装

二、整体架构

2.1 PMP 插件的组成

```
PMP 插件 (pmp.ko)
├── 协议处理层
│   ├── pmp_ipc_handler()       ← IPC 消息分发

│   ├── pmp_wait()              ← 等待进程
│   ├── pmp_kill()              ← 终止进程
│   ├── pmp_signal()            ← 发送信号
│   ├── pmp_sched_set()         ← 设置调度策略

│   ├── pmp_get_process_list()  ← 获取进程列表
│   ├── pmp_get_process_info()  ← 获取进程信息
│   └── pmp_set_affinity()      ← 设置 CPU 亲和性
│
├── 进程管理
│   ├── pmp_fork.c              ← fork/clone 实现
│   ├── pmp_exec.c              ← exec 实现
│   ├── pmp_wait.c              ← wait 实现
│   └── pmp_kill.c              ← kill 实现
│
├── 信号管理
│   └── pmp_signal.c            ← 信号发送/处理
│
├── 调度管理
│   └── pmp_sched.c             ← 调度策略管理
│
└── 初始化/清理
    ├── pmp_init()              ← 模块加载
    └── pmp_exit()              ← 模块卸载
```

2.2 PMP 在 HWRun OS 中的位置

```
系统启动顺序：
BIOS/UEFI → 引导加载器 → 微内核 → METAPROTO → BUS → PARAM → LOG → GIT → HAP → PMP → FSP → NP → SP → 其他插件
                                                                                    ↑
                                                                           PMP 在这里加载
```

PMP 是第二个加载的硬件相关插件。在它之后，依赖进程管理的插件（SCHED、SHELL、CLUSTER 等）才能正常工作。

2.3 PMP 的上层消费者

上层插件 需要 PMP 提供什么
SCHED (调度器) 获取进程列表、设置调度策略和优先级
SHELL fork/exec 执行命令、wait 等待命令完成
CLUSTER 远程进程创建和管理
SANDBOX 进程隔离、权限控制
CONTAINER 容器内进程管理
MONITOR 获取进程列表和状态信息

2.4 关键内核概念

Linux 内核中，进程和线程都由 task_struct 结构体描述：

概念 内核表示 说明
进程 task_struct 每个进程一个 task_struct
线程 task_struct 线程也是 task_struct，共享 mm_struct
进程组 task_struct->tgid 线程组 ID
父进程 task_struct->parent 父进程指针

进程状态：

· TASK_RUNNING (0)：可运行
· TASK_INTERRUPTIBLE (1)：可中断睡眠
· TASK_UNINTERRUPTIBLE (2)：不可中断睡眠

```c
/* ============================================================
   进程状态 (对应 Linux 的 task_state)

typedef enum {
    PMP_TASK_RUNNING = 0,           // 可运行
    PMP_TASK_INTERRUPTIBLE = 1,     // 可中断睡眠
    PMP_TASK_UNINTERRUPTIBLE = 2,   // 不可中断睡眠
    PMP_TASK_ZOMBIE = 3,            // 僵尸进程
    PMP_TASK_STOPPED = 4,           // 已停止
    PMP_TASK_DEAD = 5,              // 已死亡

   调度策略 (对应 Linux 的 sched policy)
   ============================================================ */

    PMP_SCHED_OTHER = 0,            // 普通分时调度

   进程信息
   ============================================================ */
```

5.2 进程创建 (Fork/Clone)

```c
// pmp/src/pmp_fork.c

#include <linux/kernel.h>
#include <linux/sched.h>
#include <linux/sched/task.h>
#include <linux/sched/signal.h>
#include <linux/slab.h>

#include "pmp.h"
#include "pmp_ipc.h"

/* ============================================================
   Fork 系统调用封装
   ============================================================ */
int pmp_fork(uint32_t flags, void *child_stack, void *ptid, void *ctid, void *newtls,
             int32_t *out_pid) {
    struct task_struct *child;
    pid_t pid;
    long ret;
    
    // 使用 kernel_clone 创建进程[reference:4]
    // kernel_clone 是 Linux 内核中所有 fork/clone/vfork 的统一入口
    // 不同调用通过 flags 参数区分功能[reference:5]
    ret = kernel_clone((unsigned long)flags, (unsigned long)child_stack,
                       (unsigned long)ptid, (unsigned long)ctid,
                       (unsigned long)newtls);
    
    if (ret < 0) {
        *out_pid = -1;
        return ret;
    }
    
    *out_pid = ret;
    return 0;
}

/* ============================================================
   Clone 系统调用封装 (与 fork 相同，但语义不同)
   ============================================================ */
int pmp_clone(uint32_t flags, void *child_stack, void *ptid, void *ctid, void *newtls,
              int32_t *out_pid) {
    // clone 与 fork 在底层都是调用 kernel_clone
    // 区别在于 flags 参数[reference:6]
    return pmp_fork(flags, child_stack, ptid, ctid, newtls, out_pid);
}

/* ============================================================
   Fork IPC 处理器
   ============================================================ */
void pmp_ipc_fork_handler(uint64_t from_task, void *data, size_t len) {
    pmp_fork_req_t *req = (pmp_fork_req_t *)data;
    pmp_fork_resp_t resp;
    int ret;
    
    ret = pmp_fork(req->flags, req->child_stack, req->ptid, req->ctid, req->newtls,
                   &resp.pid);
    
    resp.error = ret;
    ipc_reply(from_task, &resp, sizeof(resp));
}
```

5.3 程序执行 (Exec)

```c
// pmp/src/pmp_exec.c

#include <linux/kernel.h>
#include <linux/sched.h>
#include <linux/binfmts.h>
#include <linux/fs.h>
#include <linux/mm.h>
#include <linux/slab.h>

#include "pmp.h"
#include "pmp_ipc.h"

/* ============================================================
   Exec 系统调用封装
   ============================================================ */
int pmp_exec(const char *path, char **argv, char **envp) {
    struct file *file;
    struct linux_binprm *bprm;
    int ret;
    
    // 1. 打开可执行文件
    file = filp_open(path, O_RDONLY, 0);
    if (IS_ERR(file)) {
        return PTR_ERR(file);
    }
    
    // 2. 分配 bprm 结构
    bprm = kzalloc(sizeof(struct linux_binprm), GFP_KERNEL);
    if (!bprm) {
        filp_close(file, NULL);
        return -ENOMEM;
    }
    
    // 3. 初始化 bprm
    bprm->file = file;
    bprm->filename = (char *)path;
    bprm->interp = (char *)path;
    bprm->argc = 0;
    bprm->envc = 0;
    
    // 4. 设置参数和环境变量 (简化版)
    if (argv) {
        while (argv[bprm->argc]) bprm->argc++;
    }
    if (envp) {
        while (envp[bprm->envc]) bprm->envc++;
    }
    
    // 5. 执行程序
    ret = bprm_execve(bprm);
    
    // 6. 清理 (如果 exec 失败)
    if (ret < 0) {
        kfree(bprm);
        filp_close(file, NULL);
    }
    
    return ret;
}

/* ============================================================
   Exec IPC 处理器
   ============================================================ */
void pmp_ipc_exec_handler(uint64_t from_task, void *data, size_t len) {
```

```c
// pmp/src/pmp_wait.c

#include <linux/kernel.h>
#include <linux/sched.h>
#include <linux/sched/signal.h>
#include <linux/wait.h>

#include "pmp.h"
#include "pmp_ipc.h"

/* ============================================================
   Wait 系统调用封装
   ============================================================ */
int pmp_wait(int32_t pid, int32_t options, int32_t *out_status) {
    struct task_struct *p;
    int status;
    long ret;
    
    // 查找目标进程
    if (pid > 0) {
        p = find_task_by_vpid(pid);
        if (!p) {
            *out_status = 0;
            return -ECHILD;
        }
    } else {
        p = current->parent;
        // 实际实现中需要遍历所有子进程
    }
    
    // 等待进程状态变化
    // 实际实现使用 wait4 系统调用
    ret = kernel_wait4(pid, &status, options, NULL);
    
    if (ret < 0) {
        *out_status = 0;
        return ret;
    }
    
    *out_status = status;
    return ret;
}

/* ============================================================
   Wait IPC 处理器
   ============================================================ */
void pmp_ipc_wait_handler(uint64_t from_task, void *data, size_t len) {
    pmp_wait_req_t *req = (pmp_wait_req_t *)data;
    pmp_wait_resp_t resp;
    int ret;
    
    ret = pmp_wait(req->pid, req->options, &resp.status);
    resp.pid = ret;
    resp.error = (ret < 0) ? ret : 0;
    
    ipc_reply(from_task, &resp, sizeof(resp));
}
```

```c
// pmp/src/pmp_kill.c

#include <linux/kernel.h>
#include <linux/sched.h>
#include <linux/sched/signal.h>
#include <linux/signal.h>

#include "pmp.h"
#include "pmp_ipc.h"

/* ============================================================
   Kill 系统调用封装
   ============================================================ */
int pmp_kill(int32_t pid, int32_t sig) {
    struct pid *p;
    struct task_struct *task;
    int ret;
    
    if (pid > 0) {
        // 查找目标进程
        task = find_task_by_vpid(pid);
        if (!task) {
            return -ESRCH;
        }
        
        // 发送信号[reference:7]
        ret = send_signal(sig, task, 0);
    } else if (pid == 0) {
        // 发送给当前进程组的所有进程
        ret = kill_pgrp(task_pgrp(current), sig, 1);
    } else if (pid == -1) {
        // 发送给所有进程
        ret = kill_pid(find_vpid(1), sig, 1);
    } else {
        // 发送给指定进程组
        ret = kill_pgrp(find_vpid(-pid), sig, 1);
    }
    
    return ret;
}

/* ============================================================
   Kill IPC 处理器
   ============================================================ */
void pmp_ipc_kill_handler(uint64_t from_task, void *data, size_t len) {
    pmp_signal_req_t *req = (pmp_signal_req_t *)data;
    int ret;
    
    ret = pmp_kill(req->pid, req->sig);
    ipc_reply(from_task, &ret, sizeof(ret));
}
```

5.6 进程列表和进程信息

```c
// pmp/src/pmp_list.c

#include <linux/kernel.h>
#include <linux/sched.h>
#include <linux/sched/signal.h>
#include <linux/mm.h>
#include <linux/slab.h>

#include "pmp.h"
#include "pmp_ipc.h"

/* ============================================================
   获取进程列表
   ============================================================ */
int pmp_get_process_list(pmp_process_info_t **out_list, uint32_t *out_count) {
    struct task_struct *p;
    pmp_process_info_t *list;
    uint32_t count = 0;
    
    // 先计数
    rcu_read_lock();
    for_each_process(p) {
        count++;
    }
    rcu_read_unlock();
    
    if (count == 0) {
        *out_list = NULL;
        *out_count = 0;
        return 0;
    }
    
    // 分配内存
    list = kmalloc(count * sizeof(pmp_process_info_t), GFP_KERNEL);
    if (!list) {
        return -ENOMEM;
    }
    
    // 填充信息
    uint32_t idx = 0;
    rcu_read_lock();
    for_each_process(p) {
        pmp_process_info_t *info = &list[idx];
        
        info->pid = task_pid_nr(p);
        info->ppid = task_ppid_nr(p);
        info->tgid = task_tgid_nr(p);
        strncpy(info->comm, p->comm, sizeof(info->comm) - 1);
        info->state = p->__state;
        info->priority = p->prio;
        info->nice = task_nice(p);
        info->policy = p->policy;
        
        info->cpu_time_user = p->utime;
        info->cpu_time_system = p->stime;
        
        // CPU 亲和性
        info->cpu_affinity = 0;
        if (p->cpus_ptr) {
            // 简化版：只取第一个 CPU
            info->cpu_affinity = cpumask_first(p->cpus_ptr);
        }
        
        // 内存信息
        if (p->mm) {
            info->memory_rss = get_mm_rss(p->mm) * PAGE_SIZE / 1024;
            info->memory_virtual = p->mm->total_vm * PAGE_SIZE / 1024;
        } else {
            info->memory_rss = 0;
            info->memory_virtual = 0;
        }
        
        info->numa_node = 0;  // 简化版
        info->start_time = p->start_time;
        
        idx++;
    }
    rcu_read_unlock();
    
    *out_list = list;
    *out_count = count;
    return 0;
}

/* ============================================================
   获取单个进程信息
   ============================================================ */
int pmp_get_process_info(int32_t pid, pmp_process_info_t *out_info) {
    struct task_struct *p;
    
    if (pid <= 0) {
        return -EINVAL;
    }
    
    rcu_read_lock();
    p = find_task_by_vpid(pid);
    if (!p) {
        rcu_read_unlock();
        return -ESRCH;
    }
    
    out_info->pid = task_pid_nr(p);
    out_info->ppid = task_ppid_nr(p);
    out_info->tgid = task_tgid_nr(p);
    strncpy(out_info->comm, p->comm, sizeof(out_info->comm) - 1);
    out_info->state = p->__state;
    out_info->priority = p->prio;
    out_info->nice = task_nice(p);
    out_info->policy = p->policy;
    out_info->cpu_time_user = p->utime;
    out_info->cpu_time_system = p->stime;
    
    out_info->cpu_affinity = 0;
    if (p->cpus_ptr) {
        out_info->cpu_affinity = cpumask_first(p->cpus_ptr);
    }
    
    if (p->mm) {
        out_info->memory_rss = get_mm_rss(p->mm) * PAGE_SIZE / 1024;
        out_info->memory_virtual = p->mm->total_vm * PAGE_SIZE / 1024;
    } else {
        out_info->memory_rss = 0;
        out_info->memory_virtual = 0;
    }
    
    out_info->numa_node = 0;
    out_info->start_time = p->start_time;
    
    rcu_read_unlock();
    return 0;
}
```

```c
    int ret;
    
    if (pid == 0) {
```

5.8 亲和性管理

```c
// pmp/src/pmp_affinity.c

#include <linux/kernel.h>
#include <linux/sched.h>
#include <linux/sched/rt.h>
#include <linux/cpumask.h>

#include "pmp.h"
#include "pmp_ipc.h"

/* ============================================================
   设置 CPU 亲和性
   ============================================================ */
int pmp_set_affinity(int32_t pid, uint32_t cpu_mask) {
    struct task_struct *p;
    cpumask_t new_mask;
    
    if (pid == 0) {
        p = current;
    } else {
        p = find_task_by_vpid(pid);
        if (!p) {
            return -ESRCH;
        }
    }
    
    // 构建 CPU 掩码
    cpumask_clear(&new_mask);
    for (int cpu = 0; cpu < NR_CPUS; cpu++) {
        if (cpu_mask & (1 << cpu)) {
            cpumask_set_cpu(cpu, &new_mask);
        }
    }
    
    // 设置亲和性
    return sched_setaffinity(p->pid, &new_mask);
}

/* ============================================================
   获取 CPU 亲和性
   ============================================================ */
int pmp_get_affinity(int32_t pid, uint32_t *out_cpu_mask) {
    struct task_struct *p;
    cpumask_t mask;
    uint32_t cpu_mask = 0;
    
    if (pid == 0) {
        p = current;
    } else {
        p = find_task_by_vpid(pid);
        if (!p) {
            return -ESRCH;
        }
    }
    
    // 获取亲和性
    sched_getaffinity(p->pid, &mask);
    
    for (int cpu = 0; cpu < NR_CPUS; cpu++) {
        if (cpumask_test_cpu(cpu, &mask)) {
            cpu_mask |= (1 << cpu);
        }
    }
    
    *out_cpu_mask = cpu_mask;
    return 0;
}
```

5.9 IPC 消息处理

```c
// pmp/src/pmp_ipc.c

#include <linux/slab.h>
#include <linux/uaccess.h>

#include "pmp.h"
#include "pmp_ipc.h"

/* ============================================================
   IPC 消息处理入口
   ============================================================ */
void pmp_ipc_handler(uint64_t from_task, void *data, size_t len) {
    uint32_t msg_type = *(uint32_t *)data;
    
    switch (msg_type) {
        case PMP_MSG_FORK:
            pmp_ipc_fork_handler(from_task, data, len);
            break;
            
        case PMP_MSG_EXEC:
            pmp_ipc_exec_handler(from_task, data, len);
            break;
            
        case PMP_MSG_WAIT:
            pmp_ipc_wait_handler(from_task, data, len);
            break;
            
        case PMP_MSG_KILL:
            pmp_ipc_kill_handler(from_task, data, len);
            break;
            
        case PMP_MSG_SIGNAL:
            pmp_ipc_signal_handler(from_task, data, len);
            break;
            
        case PMP_MSG_SCHED_SET:
            pmp_ipc_sched_set_handler(from_task, data, len);
            break;
            
        case PMP_MSG_SCHED_GET:
            pmp_ipc_sched_get_handler(from_task, data, len);
            break;
            
        case PMP_MSG_GET_PROCESS_LIST: {
            pmp_process_info_t *list;
            uint32_t count;
            int ret = pmp_get_process_list(&list, &count);
            if (ret == 0) {
                ipc_reply(from_task, list, count * sizeof(pmp_process_info_t));
                if (list) kfree(list);
            } else {
                ipc_reply(from_task, NULL, 0);
            }
            break;
        }
        
        case PMP_MSG_GET_PROCESS_INFO: {
            int32_t pid = *(int32_t *)data;
            pmp_process_info_t info;
            int ret = pmp_get_process_info(pid, &info);
            if (ret == 0) {
                ipc_reply(from_task, &info, sizeof(info));
            } else {
                ipc_reply(from_task, NULL, 0);
            }
            break;
        }
        
        case PMP_MSG_SET_AFFINITY: {
            // 处理亲和性设置
            break;
        }
        
        default:
            printk(KERN_WARNING "HWRun PMP: Unknown IPC message type: %u\n", msg_type);
            break;
    }
}
```

八、总结

维度 决策
定位 进程管理协议，所有进程相关操作的统一入口
实现方式 Linux 内核模块 (.ko)
代码量 ~1500-2000 行 C
提供协议 PMP (v1.0)
依赖 微内核 IPC + METAPROTO + PARAM + LOG + HAP
加载时机 在 SCHED、SHELL、CLUSTER 等依赖进程管理的插件之前
关键功能 fork/clone/exec/wait/kill/signal/sched_set/亲和性

PMP 是 HWRun OS 中第二个加载的硬件相关插件（HAP → PMP）。它作为内核模块运行，直接操作 Linux 内核的 task_struct，通过 PMP 协议向上层插件提供进程创建、控制、调度、信号等完整的进程管理功能。上层插件（SCHED、SHELL、CLUSTER、SANDBOX 等）通过 PMP 协议管理进程，不直接操作内核进程管理接口。
