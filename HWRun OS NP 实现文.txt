HWRun OS NP 实现文档

模块名称： NP (Network Protocol)
版本： 1.0.0
状态： 设计冻结
定位： 网络协议，提供对 socket、TCP/UDP、网络接口等网络功能的统一抽象
实现语言： C（内核模块）
依赖： 微内核（任务、IPC、内存）、METAPROTO（协议注册）、PARAM（参数读取）、LOG（日志记录）、HAP（硬件抽象）、PMP（进程管理）
提供协议： NP

一、设计哲学

1.1 核心定位

NP 是 HWRun OS 的网络核心——所有与 socket、TCP/UDP、网络接口相关的操作都通过 NP 协议进行。

```
NP = 网络协议 = 所有网络操作 (HAP) 的统一入口

上层插件 (CLUSTER, FS_TRANSFER, NODE_DISCOVERY, ...)
    │
    │  通过 NP 协议进行网络通信
    ▼
NP 插件 (内核模块)
    │
    │  直接操作内核网络栈
    ▼
Linux 内核 (Socket / TCP/IP 协议栈)
```

1.2 设计原则

1. 协议驱动：所有网络操作通过 NP 协议进行
2. 内核实现：NP 插件运行在内核态，直接操作 socket
3. 标准接口：暴露 socket/bind/listen/accept/connect/send/recv 等标准 BSD socket 接口
4. 多协议支持：支持 TCP、UDP、RAW、UNIX 域 socket
5. 异步通知：支持非阻塞和事件驱动
6. 多网络接口：支持多个网络接口的管理

1.3 NP 插件的本质

NP 插件 = Linux 内核网络栈的 HWRun 协议封装

2.2 NP 在 HWRun OS 中的位置

```
系统启动顺序：
BIOS/UEFI → 引导加载器 → 微内核 → METAPROTO → BUS → PARAM → LOG → GIT → HAP → PMP → FSP → NP → SP → 其他插件
                                                                                    ↑
                                                                           NP 在这里加载
```

NP 是第四个加载的硬件相关插件。在它之后，依赖网络的插件（CLUSTER、FS_TRANSFER、NODE_DISCOVERY 等）才能正常工作。

2.3 NP 的上层消费者

上层插件 需要 NP 提供什么
CLUSTER (集群管理) 节点间通信、心跳
FS_TRANSFER (文件传输) 跨节点文件传输
NODE_DISCOVERY (节点发现) UDP 广播/mDNS
CONSENSUS (共识协议) Raft/Paxos 网络通信
CROSS_REGION (跨地域) 中继连接

```c
   模块加载入口
   ============================================================ */
static int __init np_init(void) {
    printk(KERN_INFO "HWRun NP: Initializing...\n");
    
    // 1. 初始化上下文
    memset(&g_np_ctx, 0, sizeof(np_context_t));
    
    // 2. 获取网络接口列表

    np_register_protocol();
    
    g_np_ctx.initialized = 1;
    printk(KERN_INFO "HWRun NP: Initialized successfully\n");
    
    return 0;
}

    printk(KERN_INFO "HWRun NP: Exiting...\n");
    
    np_ipc_unregister();
    
    // 清理 socket 列表
    // 清理接口列表
    // 清理路由表
    
    g_np_ctx.initialized = 0;
    printk(KERN_INFO "HWRun NP: Exited\n");
}

module_init(np_init);
module_exit(np_exit);
```

5.2 Socket 操作

```c
// np/src/np_socket.c

#include <linux/kernel.h>
#include <linux/socket.h>
#include <linux/net.h>
#include <linux/in.h>
#include <linux/ip.h>
#include <linux/tcp.h>
#include <linux/udp.h>
#include <net/sock.h>

#include "np.h"
#include "np_ipc.h"

/* ============================================================
   创建 Socket
   ============================================================ */
int np_socket(int32_t domain, int32_t type, int32_t protocol, int32_t *out_fd) {
    struct socket *sock;
    int ret;
    
    ret = sock_create(domain, type, protocol, &sock);
    if (ret < 0) {
        *out_fd = -1;
        return ret;
    }
    
    // 获取文件描述符
    ret = sock_map_fd(sock, 0);
    if (ret < 0) {
        sock_release(sock);
        *out_fd = -1;
        return ret;
    }
    
    *out_fd = ret;
    return 0;
}

/* ============================================================
   Bind 绑定地址
   ============================================================ */
int np_bind(int32_t fd, uint32_t ip, uint32_t port) {
    struct socket *sock;
    struct sockaddr_in addr;
    int ret;
    
    // 获取 socket
    sock = sockfd_lookup(fd, &ret);
    if (!sock) {
        return ret;
    }
    
    // 构建地址
    memset(&addr, 0, sizeof(addr));
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = ip;
    addr.sin_port = htons(port);
    
    ret = kernel_bind(sock, (struct sockaddr *)&addr, sizeof(addr));
    
    sockfd_put(sock);
    return ret;
}

/* ============================================================
   Listen 监听
   ============================================================ */
int np_listen(int32_t fd, int32_t backlog) {
    struct socket *sock;
    int ret;
    
    sock = sockfd_lookup(fd, &ret);
    if (!sock) {
        return ret;
    }
    
    ret = kernel_listen(sock, backlog);
    
    sockfd_put(sock);
    return ret;
}

/* ============================================================
   Accept 接受连接
   ============================================================ */
int np_accept(int32_t fd, uint32_t *out_remote_ip, uint32_t *out_remote_port,
              int32_t *out_new_fd) {
    struct socket *sock, *newsock;
    struct sockaddr_in addr;
    int addr_len = sizeof(addr);
    int ret;
    
    sock = sockfd_lookup(fd, &ret);
    if (!sock) {
        return ret;
    }
    
    ret = kernel_accept(sock, &newsock, 0);
    if (ret < 0) {
        sockfd_put(sock);
        return ret;
    }
    
    // 获取远程地址
    ret = kernel_getpeername(newsock, (struct sockaddr *)&addr, &addr_len);
    if (ret < 0) {
        sock_release(newsock);
        sockfd_put(sock);
        return ret;
    }
    
    *out_remote_ip = addr.sin_addr.s_addr;
    *out_remote_port = ntohs(addr.sin_port);
    
    // 获取新 socket 的文件描述符
    ret = sock_map_fd(newsock, 0);
    if (ret < 0) {
        sock_release(newsock);
        sockfd_put(sock);
        return ret;
    }
    
    *out_new_fd = ret;
    
    sockfd_put(sock);
    return 0;
}

/* ============================================================
   Connect 连接服务器
   ============================================================ */
int np_connect(int32_t fd, uint32_t ip, uint32_t port) {
    struct socket *sock;
    struct sockaddr_in addr;
    int ret;
    
    sock = sockfd_lookup(fd, &ret);
    if (!sock) {
        return ret;
    }
    
    memset(&addr, 0, sizeof(addr));
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = ip;
    addr.sin_port = htons(port);
    
    ret = kernel_connect(sock, (struct sockaddr *)&addr, sizeof(addr), 0);
    
    sockfd_put(sock);
    return ret;
}

/* ============================================================
   Send 发送数据
   ============================================================ */
int np_send(int32_t fd, const void *data, uint32_t len, int32_t flags,
            uint64_t *out_sent) {
    struct socket *sock;
    struct msghdr msg;
    struct kvec iov;
    int ret;
    
    sock = sockfd_lookup(fd, &ret);
    if (!sock) {
        return ret;
    }
    
    memset(&msg, 0, sizeof(msg));
    iov.iov_base = (void *)data;
    iov.iov_len = len;
    
    ret = kernel_sendmsg(sock, &msg, &iov, 1, len);
    
    sockfd_put(sock);
    
    if (ret < 0) {
        *out_sent = 0;
        return ret;
    }
    
    *out_sent = ret;
    return 0;
}

/* ============================================================
   Recv 接收数据
   ============================================================ */
int np_recv(int32_t fd, void *buf, uint32_t max_len, int32_t flags,
            uint64_t *out_received, uint32_t *out_remote_ip, uint32_t *out_remote_port) {
    struct socket *sock;
    struct msghdr msg;
    struct kvec iov;
    struct sockaddr_in addr;
    int addr_len = sizeof(addr);
    int ret;
    
    sock = sockfd_lookup(fd, &ret);
    if (!sock) {
        return ret;
    }
    
    memset(&msg, 0, sizeof(msg));
    msg.msg_name = &addr;
    msg.msg_namelen = addr_len;
    msg.msg_flags = flags;
    
    iov.iov_base = buf;
    iov.iov_len = max_len;
    
    ret = kernel_recvmsg(sock, &msg, &iov, 1, max_len, msg.msg_flags);
    
    if (out_remote_ip && addr.sin_family == AF_INET) {
        *out_remote_ip = addr.sin_addr.s_addr;
        *out_remote_port = ntohs(addr.sin_port);
    }
    
    sockfd_put(sock);
    
    if (ret < 0) {
        *out_received = 0;
        return ret;
    }
    
    *out_received = ret;
    return 0;
}

/* ============================================================
   Close 关闭 Socket
   ============================================================ */
int np_close(int32_t fd) {
    struct socket *sock;
    int ret;
    
    sock = sockfd_lookup(fd, &ret);
    if (!sock) {
        return ret;
    }
    
    ret = sock_release(sock);
    
    return ret;
}
```

```c
// np/src/np_interface.c

#include <linux/kernel.h>
#include <linux/netdevice.h>
#include <linux/inetdevice.h>
#include <linux/if.h>
#include <linux/rtnetlink.h>
#include <net/ip.h>
#include <net/route.h>

#include "np.h"
#include "np_ipc.h"

/* ============================================================
   获取所有网络接口
   ============================================================ */
int np_interface_probe(np_interface_t **out_interfaces) {
    struct net_device *dev;
    np_interface_t *list = NULL;
    int count = 0;
    
    rtnl_lock();
    
    for_each_netdev(&init_net, dev) {
        np_interface_t *iface = kmalloc(sizeof(np_interface_t), GFP_KERNEL);
        if (!iface) {
            rtnl_unlock();
            return -ENOMEM;
        }
        
        memset(iface, 0, sizeof(np_interface_t));
        
        // 名称
        strncpy(iface->name, dev->name, sizeof(iface->name) - 1);
        
        // MAC 地址
        if (dev->dev_addr) {
            memcpy(iface->mac, dev->dev_addr, 6);
        }
        
        // IP 地址
        struct in_device *in_dev = dev->ip_ptr;
        if (in_dev) {
            struct in_ifaddr *ifa = in_dev->ifa_list;
            if (ifa) {
                iface->ipv4 = ifa->ifa_address;
                iface->netmask = ifa->ifa_mask;
                iface->gateway = 0;  // 需要从路由表获取
            }
        }
        
        // MTU
        iface->mtu = dev->mtu;
        
        // 状态
        iface->up = (dev->flags & IFF_UP) ? 1 : 0;
        
        // 统计
        iface->rx_bytes = dev->stats.rx_bytes;
        iface->tx_bytes = dev->stats.tx_bytes;
        
        iface->next = list;
        list = iface;
        count++;
    }
    
    rtnl_unlock();
    
    *out_interfaces = list;
    return count;
}

/* ============================================================
   设置 MTU
   ============================================================ */
int np_set_mtu(const char *iface_name, uint32_t mtu) {
    struct net_device *dev;
    int ret;
    
    rtnl_lock();
    
    dev = __dev_get_by_name(&init_net, iface_name);
    if (!dev) {
        rtnl_unlock();
        return -ENODEV;
    }
    
    ret = dev_set_mtu(dev, mtu);
    
    rtnl_unlock();
    return ret;
}

/* ============================================================
   设置接口状态 (up/down)
   ============================================================ */
int np_set_interface_up(const char *iface_name, int up) {
    struct net_device *dev;
    int ret;
    
    rtnl_lock();
    
    dev = __dev_get_by_name(&init_net, iface_name);
    if (!dev) {
        rtnl_unlock();
        return -ENODEV;
    }
    
    if (up) {
        ret = dev_open(dev, NULL);
    } else {
        ret = dev_close(dev);
    }
    
    rtnl_unlock();
    return ret;
}
```

```c
   添加路由

int np_add_route(uint32_t dest, uint32_t mask, uint32_t gateway,
                 uint32_t metric, const char *iface) {
    struct rtentry rt;
    struct sockaddr_in *sin;
    int ret;
    struct socket *sock;
    mm_segment_t oldfs;
    
    // 创建 socket 用于 ioctl
    ret = sock_create(AF_INET, SOCK_DGRAM, 0, &sock);
    if (ret < 0) {
        return ret;
    }
    
    memset(&rt, 0, sizeof(rt));
    
    // 目标网络
    sin = (struct sockaddr_in *)&rt.rt_dst;
    sin->sin_family = AF_INET;
    sin->sin_addr.s_addr = dest;
    
    // 子网掩码
    sin = (struct sockaddr_in *)&rt.rt_genmask;
    sin->sin_family = AF_INET;
    sin->sin_addr.s_addr = mask;
    
    // 网关
    sin = (struct sockaddr_in *)&rt.rt_gateway;
    sin->sin_family = AF_INET;
    sin->sin_addr.s_addr = gateway;
    
    rt.rt_metric = metric;
    rt.rt_flags = RTF_UP | RTF_GATEWAY;
    strncpy(rt.rt_dev, iface, IFNAMSIZ - 1);
    
    oldfs = get_fs();
    set_fs(KERNEL_DS);
    ret = sock->ops->ioctl(sock, SIOCADDRT, (unsigned long)&rt);
    set_fs(oldfs);
    
    sock_release(sock);
    return ret;
}
```

```c
#include <linux/uaccess.h>

#include "np.h"
#include "np_ipc.h"

/* ============================================================
   IPC 消息处理入口
   ============================================================ */
void np_ipc_handler(uint64_t from_task, void *data, size_t len) {
    uint32_t msg_type = *(uint32_t *)data;
    
    switch (msg_type) {
        case NP_MSG_SOCKET: {
            np_socket_req_t *req = (np_socket_req_t *)data;
            np_socket_resp_t resp;
            int ret = np_socket(req->domain, req->type, req->protocol, &resp.fd);
            resp.error = ret;
            ipc_reply(from_task, &resp, sizeof(resp));
            break;
        }
        
        case NP_MSG_BIND: {
            np_bind_req_t *req = (np_bind_req_t *)data;
            int ret = np_bind(req->fd, req->ip, req->port);
            ipc_reply(from_task, &ret, sizeof(ret));
            break;
        }
        
        case NP_MSG_LISTEN: {
            np_listen_req_t *req = (np_listen_req_t *)data;
            int ret = np_listen(req->fd, req->backlog);
            ipc_reply(from_task, &ret, sizeof(ret));
            break;
        }
        
        case NP_MSG_ACCEPT: {
            np_accept_req_t *req = (np_accept_req_t *)data;
            np_accept_resp_t resp;
            int ret = np_accept(req->fd, &resp.remote_ip, &resp.remote_port, &resp.new_fd);
            resp.error = ret;
            ipc_reply(from_task, &resp, sizeof(resp));
            break;
        }
        
        case NP_MSG_CONNECT: {
            np_connect_req_t *req = (np_connect_req_t *)data;
            int ret = np_connect(req->fd, req->ip, req->port);
            ipc_reply(from_task, &ret, sizeof(ret));
            break;
        }
        
        case NP_MSG_SEND: {
            // 处理 send
            break;
        }
        
        case NP_MSG_RECV: {
            // 处理 recv
            break;
        }
        
        case NP_MSG_CLOSE: {
            int32_t fd = *(int32_t *)data;
            int ret = np_close(fd);
            ipc_reply(from_task, &ret, sizeof(ret));
            break;
        }
        
        case NP_MSG_GET_INTERFACE: {
            // 获取网络接口信息
            break;
        }
        
        default:
            printk(KERN_WARNING "HWRun NP: Unknown IPC message type: %u\n", msg_type);
            break;
    }
}
```

八、总结

维度 决策
定位 网络协议，所有网络操作的统一入口
实现方式 Linux 内核模块 (.ko)
代码量 ~1500-2000 行 C
提供协议 NP (v1.0)
依赖 微内核 IPC + METAPROTO + PARAM + LOG + HAP + PMP
加载时机 在 CLUSTER、FS_TRANSFER、NODE_DISCOVERY 等依赖网络的插件之前
关键功能 socket/bind/listen/accept/connect/send/recv/close/接口管理/路由管理/统计

NP 是 HWRun OS 中第四个加载的硬件相关插件（HAP → PMP → FSP → NP）。它作为内核模块运行，直接操作 Linux 内核的网络栈，通过 NP 协议向上层插件提供完整的网络通信功能。上层插件（CLUSTER、FS_TRANSFER、NODE_DISCOVERY 等）通过 NP 协议进行网络通信，不直接操作内核网络栈。
