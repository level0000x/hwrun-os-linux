HWRun OS LOG 实现文档

模块名称： LOG
版本： 1.0.0
状态： 设计冻结
定位： 日志系统，所有日志的统一管理入口
实现语言： C
依赖： 微内核（任务、IPC、内存）、METAPROTO（协议注册）、PARAM（参数读取）
提供协议： LOG

一、设计哲学

1.1 核心定位

LOG 是 HWRun OS 的“日志中心”——所有插件通过它记录日志，所有日志统一存储、统一查询、统一管理。

```
BUS 启动后加载的第二个功能性插件（PARAM → LOG）
    │
    ▼
LOG (日志系统)
    │  提供统一的日志服务：
    │  1. 异步日志写入（不阻塞业务）
    │  2. 日志级别过滤（debug/info/warn/error/fatal）
    │  3. 多输出目标（文件、控制台、串口）
    │  4. 日志轮转（按大小自动轮转）
    │  5. 审计日志（关键操作记录）
    │  6. 日志查询（按时间/级别/插件过滤）
    ▼
其他插件 (GIT, HAP, FSP, NP, ...)
    │  通过 LOG 记录日志
```

1.2 设计原则

1. 异步写入：日志写入使用异步队列，不阻塞业务线程
2. 线程安全：支持多线程并发写入
3. 级别过滤：支持 debug/info/warn/error/fatal 五级
4. 多输出目标：支持文件、控制台、串口同时输出
5. 自动轮转：按大小自动轮转，防止磁盘占满
6. 审计日志：单独文件记录关键操作
7. 可配置：通过 PARAM 系统配置日志行为

1.3 日志级别

级别 数值 用途 示例
DEBUG 0 调试信息 DEBUG: scheduler.c:123 policy changed
INFO 1 一般信息 INFO: Plugin linux-kernel loaded
WARN 2 警告 WARN: Memory usage high 85%
ERROR 3 错误 ERROR: Failed to open file /etc/hwrun/config
FATAL 4 致命错误 FATAL: Kernel panic - cannot recover

过滤规则：只记录级别 ≥ 当前设置级别的日志（默认 INFO）。

三、核心数据结构

```c
// log/include/log.h

#include <stdint.h>
#include <stddef.h>
#include <time.h>

/* ============================================================
   日志级别
   ============================================================ */
typedef enum {
    LOG_LEVEL_DEBUG = 0,
    LOG_LEVEL_INFO = 1,
    LOG_LEVEL_WARN = 2,
    LOG_LEVEL_ERROR = 3,
    LOG_LEVEL_FATAL = 4,
    LOG_LEVEL_NONE = 5,             // 禁用日志
} log_level_t;

/* ============================================================
   日志消息
   ============================================================ */
#define LOG_MSG_MAX 1024
#define LOG_PLUGIN_MAX 64
#define LOG_FILE_MAX 128

typedef struct log_message {
    uint64_t timestamp;                 // 时间戳（微秒）
    log_level_t level;                  // 日志级别
    char plugin[LOG_PLUGIN_MAX];        // 插件 ID
    char file[LOG_FILE_MAX];            // 源文件
    int line;                           // 行号
    char msg[LOG_MSG_MAX];              // 消息内容
    struct log_message *next;           // 队列链表
} log_message_t;

/* ============================================================
   输出目标类型
   ============================================================ */
typedef enum {
    LOG_OUTPUT_FILE = 0,                // 文件
    LOG_OUTPUT_CONSOLE,                  // 控制台
    LOG_OUTPUT_SERIAL,                   // 串口
} log_output_type_t;

/* ============================================================
   输出目标
   ============================================================ */
typedef struct log_output {
    log_output_type_t type;
    int enabled;
    log_level_t min_level;              // 该输出的最低级别
    
    union {
        struct {
            char path[256];             // 文件路径
            int fd;                     // 文件描述符
            uint64_t current_size;      // 当前文件大小
            uint64_t max_size;          // 最大大小（触发轮转）
            int max_files;              // 保留历史文件数
        } file;
        struct {
            int fd;                     // stdout/stderr
        } console;
        struct {
            char port[32];              // 串口设备
            int baud;                   // 波特率
            int fd;                     // 串口文件描述符
        } serial;
    } config;
    struct log_output *next;
} log_output_t;

/* ============================================================
   日志查询请求
   ============================================================ */
typedef struct log_query {
    log_level_t min_level;              // 最低级别
    uint64_t start_time;                // 开始时间
    uint64_t end_time;                  // 结束时间
    char plugin[LOG_PLUGIN_MAX];        // 插件过滤（空表示全部）
    char keyword[64];                   // 关键词搜索
    int limit;                          // 返回条数限制
    int offset;                         // 偏移量
} log_query_t;

/* ============================================================
   LOG 上下文
   ============================================================ */
#define LOG_QUEUE_SIZE 4096              // 队列大小（消息数）

typedef struct log_context {
    /* 队列 */
    log_message_t *queue_head;          // 队列头
    log_message_t *queue_tail;          // 队列尾
    uint32_t queue_count;               // 当前队列长度
    uint32_t queue_max;                 // 最大队列长度
    
    /* 输出目标 */
    log_output_t *outputs;              // 输出目标链表
    
    /* 写入线程 */
    uint64_t writer_task_id;            // 写入线程任务 ID
    int writer_running;                 // 写入线程运行标志
    
    /* 配置 */
    log_level_t global_min_level;       // 全局最低级别
    int audit_enabled;                  // 审计日志是否启用
    
    /* 持久化路径 */
    char log_dir[256];
    char main_log_path[256];
    char audit_log_path[256];
    
    /* 互斥锁 */
    void *queue_lock;                   // 队列锁
    void *file_lock;                    // 文件写入锁
    
    /* 状态 */
    int initialized;
    uint64_t log_task_id;               // LOG 自身的任务 ID
} log_context_t;

/* ============================================================
   LOG 协议消息结构（通过 IPC 调用）
   ============================================================ */

/* 写入日志请求 */
typedef struct log_write_req {
    log_level_t level;
    char plugin[LOG_PLUGIN_MAX];
    char file[LOG_FILE_MAX];
    int line;
    char msg[LOG_MSG_MAX];
} log_write_req_t;

/* 写入日志响应 */
typedef struct log_write_resp {
    int success;
    int dropped;                        // 是否因队列满被丢弃
} log_write_resp_t;

/* 查询日志请求 */
typedef struct log_query_req {
    log_query_t query;
} log_query_req_t;

/* 查询日志响应 */
typedef struct log_query_resp {
    int total;
    int returned;
    char *messages;                     // 多条消息，用 \n 分隔
} log_query_resp_t;

/* 日志级别设置请求 */
typedef struct log_setlevel_req {
    log_level_t level;
} log_setlevel_req_t;
```

```protobuf
// protocols/log.proto
syntax = "proto3";

package hwrun.log;

service Logger {
    // === 日志写入 ===
    rpc Write(WriteRequest) returns (WriteResponse);
    
    // === 日志查询 ===
    rpc Read(ReadRequest) returns (ReadResponse);
    rpc Tail(TailRequest) returns (TailResponse);
    rpc Search(SearchRequest) returns (SearchResponse);
    
    // === 日志管理 ===
    rpc Rotate(RotateRequest) returns (RotateResponse);
    rpc SetLevel(SetLevelRequest) returns (SetLevelResponse);
    rpc GetLevel(Empty) returns (GetLevelResponse);
}

message WriteRequest {
    uint32 level = 1;                 // 0=DEBUG, 1=INFO, 2=WARN, 3=ERROR, 4=FATAL
    string plugin = 2;
    string file = 3;
    uint32 line = 4;
    string msg = 5;
}

message WriteResponse {
    bool success = 1;
    bool dropped = 2;
}

message ReadRequest {
    uint32 min_level = 1;
    uint64 start_time = 2;
    uint64 end_time = 3;
    string plugin = 4;
    string keyword = 5;
    uint32 limit = 6;
    uint32 offset = 7;
}

message ReadResponse {
    uint32 total = 1;
    uint32 returned = 2;
    string messages = 3;
    bool success = 4;
    string error = 5;
}

message TailRequest {
    uint32 lines = 1;                 // 最近 N 行
}

message TailResponse {
    string messages = 1;
    bool success = 2;
    string error = 3;
}

message SearchRequest {
    string keyword = 1;
    uint32 limit = 2;
}

message SearchResponse {
    string messages = 1;
    bool success = 2;
    string error = 3;
}

message RotateRequest {
    bool force = 1;                   // 强制轮转
}

message RotateResponse {
    bool success = 1;
    string error = 2;
}

message SetLevelRequest {
    uint32 level = 1;
}

message SetLevelResponse {
    bool success = 1;
    string error = 2;

message Empty {}

}
```

5.1 初始化与自举

```c
// log/src/log.c

#include <stdint.h>
#include <string.h>
#include <stdlib.h>
#include <unistd.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <time.h>

#include "log.h"
#include "kernel_ipc.h"
#include "kernel_task.h"
#include "kernel_mm.h"
#include "metaproto_client.h"
#include "param_client.h"

/* 全局 LOG 上下文 */
static log_context_t g_ctx = {0};

/* ============================================================
   LOG 初始化（被 BUS 调用）
   ============================================================ */
int log_init(uint64_t task_id) {
    memset(&g_ctx, 0, sizeof(g_ctx));
    
    // 1. 记录任务 ID
    g_ctx.log_task_id = task_id;
    
    // 2. 设置默认路径
    strcpy(g_ctx.log_dir, "/var/log/hwrun/");
    snprintf(g_ctx.main_log_path, sizeof(g_ctx.main_log_path), 
             "%s/hwrun.log", g_ctx.log_dir);
    snprintf(g_ctx.audit_log_path, sizeof(g_ctx.audit_log_path), 
             "%s/audit.log", g_ctx.log_dir);
    
    // 3. 创建日志目录
    mkdir(g_ctx.log_dir, 0755);
    
    // 4. 从 PARAM 读取配置
    char buf[64];
    if (param_get("log.level", buf, sizeof(buf)) == 0) {
        g_ctx.global_min_level = log_level_from_string(buf);
    } else {
        g_ctx.global_min_level = LOG_LEVEL_INFO;  // 默认 INFO
    }
    
    if (param_get("log.audit_enabled", buf, sizeof(buf)) == 0) {
        g_ctx.audit_enabled = (strcmp(buf, "true") == 0);
    } else {
        g_ctx.audit_enabled = 1;  // 默认开启
    }
    
    // 5. 初始化队列
    g_ctx.queue_head = NULL;
    g_ctx.queue_tail = NULL;
    g_ctx.queue_count = 0;
    g_ctx.queue_max = LOG_QUEUE_SIZE;
    
    // 6. 初始化输出目标
    log_init_outputs();
    
    // 7. 创建日志写入线程
    g_ctx.writer_running = 1;
    g_ctx.writer_task_id = task_create("log_writer", log_writer_thread, NULL);
    
    // 8. 注册 LOG 协议（通过 METAPROTO）
    ProtocolInfo proto_info = {
        .name = "LOG",
        .version = "1.0.0",
        .description = "Logging System Protocol",
        .provider_plugin = "log-system",
        .status = "active",
    };
    metaproto_register(metaproto_client, &proto_info);
    
    // 9. 注册 IPC 消息处理器
    ipc_register_handler(LOG_MSG_WRITE, log_ipc_write_handler);
    ipc_register_handler(LOG_MSG_READ, log_ipc_read_handler);
    ipc_register_handler(LOG_MSG_TAIL, log_ipc_tail_handler);
    ipc_register_handler(LOG_MSG_SETLEVEL, log_ipc_setlevel_handler);
    
    g_ctx.initialized = 1;
    log_write_internal(NULL, LOG_LEVEL_INFO, "log-system", __FILE__, __LINE__, 
                       "LOG initialized, level=%d", g_ctx.global_min_level);
    return 0;
}

/* ============================================================
   初始化输出目标
   ============================================================ */
void log_init_outputs(void) {
    char buf[64];
    
    // 1. 文件输出（默认启用）
    log_output_t *file_out = kmalloc(sizeof(log_output_t));
    if (file_out) {
        file_out->type = LOG_OUTPUT_FILE;
        file_out->enabled = 1;
        file_out->min_level = LOG_LEVEL_DEBUG;
        strcpy(file_out->config.file.path, g_ctx.main_log_path);
        file_out->config.file.fd = -1;
        file_out->config.file.max_size = 10 * 1024 * 1024;  // 10MB
        file_out->config.file.max_files = 5;
        file_out->next = g_ctx.outputs;
        g_ctx.outputs = file_out;
        
        // 打开日志文件
        file_out->config.file.fd = open(file_out->config.file.path, 
                                         O_WRONLY | O_CREAT | O_APPEND, 0644);
    }
    
    // 2. 控制台输出（默认启用）
    log_output_t *console_out = kmalloc(sizeof(log_output_t));
    if (console_out) {
        console_out->type = LOG_OUTPUT_CONSOLE;
        console_out->enabled = 1;
        console_out->min_level = LOG_LEVEL_INFO;
        console_out->config.console.fd = 1;  // stdout
        console_out->next = g_ctx.outputs;
        g_ctx.outputs = console_out;
    }
    
    // 3. 串口输出（根据配置）
    if (param_get("log.serial.enabled", buf, sizeof(buf)) == 0 && 
        strcmp(buf, "true") == 0) {
        log_output_t *serial_out = kmalloc(sizeof(log_output_t));
        if (serial_out) {
            serial_out->type = LOG_OUTPUT_SERIAL;
            serial_out->enabled = 1;
            serial_out->min_level = LOG_LEVEL_DEBUG;
            
            char port[32];
            param_get("log.serial.port", port, sizeof(port));
            strcpy(serial_out->config.serial.port, port[0] ? port : "/dev/ttyS0");
            
            char baud_buf[16];
            param_get("log.serial.baud", baud_buf, sizeof(baud_buf));
            serial_out->config.serial.baud = baud_buf[0] ? atoi(baud_buf) : 115200;
            serial_out->config.serial.fd = -1;
            
            // 打开串口
            serial_out->config.serial.fd = serial_open(
                serial_out->config.serial.port, 
                serial_out->config.serial.baud);
            
            serial_out->next = g_ctx.outputs;
            g_ctx.outputs = serial_out;
        }
    }
}
```

5.2 异步日志队列

```c
// log/src/queue.c

/* ============================================================
   将日志消息入队（非阻塞）
   返回 1 表示成功，0 表示队列满丢弃
   ============================================================ */
int log_enqueue(log_message_t *msg) {
    if (!g_ctx.initialized) return 0;
    
    // 加锁
    lock_acquire(g_ctx.queue_lock);
    
    // 检查队列是否已满
    if (g_ctx.queue_count >= g_ctx.queue_max) {
        lock_release(g_ctx.queue_lock);
        return 0;  // 丢弃
    }
    
    // 入队
    msg->next = NULL;
    if (g_ctx.queue_tail) {
        g_ctx.queue_tail->next = msg;
    } else {
        g_ctx.queue_head = msg;
    }
    g_ctx.queue_tail = msg;
    g_ctx.queue_count++;
    
    lock_release(g_ctx.queue_lock);
    return 1;
}

/* ============================================================
   从队列取出消息（阻塞）
   ============================================================ */
log_message_t *log_dequeue(void) {
    lock_acquire(g_ctx.queue_lock);
    
    if (g_ctx.queue_count == 0) {
        lock_release(g_ctx.queue_lock);
        return NULL;
    }
    
    log_message_t *msg = g_ctx.queue_head;
    g_ctx.queue_head = msg->next;
    if (!g_ctx.queue_head) {
        g_ctx.queue_tail = NULL;
    }
    g_ctx.queue_count--;
    
    lock_release(g_ctx.queue_lock);
    return msg;
}
```

5.3 日志写入线程

```c
// log/src/writer.c

/* ============================================================
   日志写入线程（消费者）
   ============================================================ */
void log_writer_thread(void *arg) {
    log_message_t *msg;
    
    while (g_ctx.writer_running) {
        // 从队列取消息（阻塞）
        msg = log_dequeue();
        
        if (msg) {
            // 写入所有输出目标
            log_write_to_outputs(msg);
            
            // 释放消息内存
            kfree(msg);
        } else {
            // 队列为空，短暂休眠
            task_sleep(10);
        }
    }
}

/* ============================================================
   写入所有输出目标
   ============================================================ */
void log_write_to_outputs(log_message_t *msg) {
    // 检查级别过滤
    if (msg->level < g_ctx.global_min_level) {
        return;
    }
    
    log_output_t *out = g_ctx.outputs;
    while (out) {
        if (out->enabled && msg->level >= out->min_level) {
            switch (out->type) {
                case LOG_OUTPUT_FILE:
                    log_write_file(out, msg);
                    break;
                case LOG_OUTPUT_CONSOLE:
                    log_write_console(out, msg);
                    break;
                case LOG_OUTPUT_SERIAL:
                    log_write_serial(out, msg);
                    break;
            }
        }
        out = out->next;
    }
    
    // 审计日志（单独处理）
    if (g_ctx.audit_enabled && 
        (msg->level == LOG_LEVEL_WARN || 
         msg->level == LOG_LEVEL_ERROR || 
         msg->level == LOG_LEVEL_FATAL)) {
        log_write_audit(msg);
    }
}

/* ============================================================
   格式化日志消息
   ============================================================ */
void log_format_message(log_message_t *msg, char *buf, size_t size) {
    char time_str[32];
    struct tm tm;
    time_t t = msg->timestamp / 1000000;
    localtime_r(&t, &tm);
    strftime(time_str, sizeof(time_str), "%Y-%m-%d %H:%M:%S", &tm);
    
    char *level_str;
    switch (msg->level) {
        case LOG_LEVEL_DEBUG: level_str = "DEBUG"; break;
        case LOG_LEVEL_INFO:  level_str = "INFO";  break;
        case LOG_LEVEL_WARN:  level_str = "WARN";  break;
        case LOG_LEVEL_ERROR: level_str = "ERROR"; break;
        case LOG_LEVEL_FATAL: level_str = "FATAL"; break;
        default: level_str = "UNKN"; break;
    }
    
    snprintf(buf, size, "[%s] [%s] [%s:%d] %s\n",
             time_str, level_str, msg->file, msg->line, msg->msg);
}
```

5.4 日志轮转

```c
// log/src/rotate.c

/* ============================================================
   检查并执行日志轮转
   ============================================================ */
int log_check_rotate(log_output_t *out) {
    if (out->type != LOG_OUTPUT_FILE) return 0;
    
    // 获取当前文件大小
    struct stat st;
    if (fstat(out->config.file.fd, &st) != 0) {
        return -1;
    }
    
    out->config.file.current_size = st.st_size;
    
    // 检查是否超过最大大小
    if (out->config.file.current_size < out->config.file.max_size) {
        return 0;  // 不需要轮转
    }
    
    // 执行轮转
    return log_rotate_file(out);
}

/* ============================================================
   执行日志文件轮转
   ============================================================ */
int log_rotate_file(log_output_t *out) {
    if (out->type != LOG_OUTPUT_FILE) return -1;
    
    char old_path[512];
    char new_path[512];
    
    // 关闭当前文件
    close(out->config.file.fd);
    
    // 轮转历史文件（从后往前移）
    for (int i = out->config.file.max_files - 1; i >= 1; i--) {
        if (i == 1) {
            snprintf(old_path, sizeof(old_path), "%s", out->config.file.path);
        } else {
            snprintf(old_path, sizeof(old_path), "%s.%d", 
                     out->config.file.path, i - 1);
        }
        snprintf(new_path, sizeof(new_path), "%s.%d", 
                 out->config.file.path, i);
        
        // 如果旧文件存在，移动到新位置
        if (access(old_path, F_OK) == 0) {
            rename(old_path, new_path);
        }
    }
    
    // 重新打开主日志文件
    out->config.file.fd = open(out->config.file.path, 
                                O_WRONLY | O_CREAT | O_APPEND, 0644);
    if (out->config.file.fd < 0) {
        return -1;
    }
    
    // 重置大小计数
    out->config.file.current_size = 0;
    
    // 记录轮转事件
    log_write_internal(NULL, LOG_LEVEL_INFO, "log-system", __FILE__, __LINE__,
                       "Log rotated, max_files=%d", out->config.file.max_files);
    
    return 0;
}
```

5.5 审计日志

```c
// log/src/audit.c

/* ============================================================
   写入审计日志
   ============================================================ */
void log_write_audit(log_message_t *msg) {
    int fd = open(g_ctx.audit_log_path, O_WRONLY | O_CREAT | O_APPEND, 0644);
    if (fd < 0) return;
    
    char buf[LOG_MSG_MAX + 128];
    char time_str[32];
    struct tm tm;
    time_t t = msg->timestamp / 1000000;
    localtime_r(&t, &tm);
    strftime(time_str, sizeof(time_str), "%Y-%m-%d %H:%M:%S", &tm);
    
    // 审计日志格式：时间 | 级别 | 插件 | 消息
    char *level_str;
    switch (msg->level) {
        case LOG_LEVEL_WARN:  level_str = "WARN";  break;
        case LOG_LEVEL_ERROR: level_str = "ERROR"; break;
        case LOG_LEVEL_FATAL: level_str = "FATAL"; break;
        default: level_str = "INFO"; break;
    }
    
    snprintf(buf, sizeof(buf), "[%s] [%s] [%s] %s\n",
             time_str, level_str, msg->plugin, msg->msg);
    
    write(fd, buf, strlen(buf));
    close(fd);
}
```

5.6 IPC 通信接口

```c
// log/src/ipc_handler.c

/* ============================================================
   日志写入 IPC 处理器
   ============================================================ */
void log_ipc_write_handler(uint64_t from_task, void *data, size_t len) {
    log_write_req_t *req = (log_write_req_t *)data;
    log_write_resp_t resp;
    
    // 分配消息
    log_message_t *msg = kmalloc(sizeof(log_message_t));
    if (!msg) {
        resp.success = 0;
        resp.dropped = 1;
        ipc_reply(from_task, &resp, sizeof(resp));
        return;
    }
    
    // 填充消息
    msg->timestamp = get_timestamp_us();
    msg->level = req->level;
    strncpy(msg->plugin, req->plugin, sizeof(msg->plugin) - 1);
    strncpy(msg->file, req->file, sizeof(msg->file) - 1);
    msg->line = req->line;
    strncpy(msg->msg, req->msg, sizeof(msg->msg) - 1);
    msg->next = NULL;
    
    // 入队
    int enqueued = log_enqueue(msg);
    
    resp.success = enqueued;
    resp.dropped = !enqueued;
    ipc_reply(from_task, &resp, sizeof(resp));
}

/* ============================================================
   日志查询 IPC 处理器
   ============================================================ */
void log_ipc_read_handler(uint64_t from_task, void *data, size_t len) {
    log_query_req_t *req = (log_query_req_t *)data;
    log_query_resp_t resp;
    
    // 从日志文件读取
    char *result = log_read_from_file(&req->query);
    
    resp.total = 0;  // 实际实现中计算总数
    resp.returned = 0;
    resp.messages = result;
    
    ipc_reply(from_task, &resp, sizeof(resp));
}

/* ============================================================
   Tail IPC 处理器
   ============================================================ */
void log_ipc_tail_handler(uint64_t from_task, void *data, size_t len) {
    uint32_t lines = *(uint32_t *)data;
    char *result = log_tail_file(lines);
    ipc_reply(from_task, result, strlen(result) + 1);
}
```

5.7 插件日志宏

```c
// log/include/log_macros.h

/* ============================================================
   插件日志宏（供其他插件使用）
   ============================================================ */

#define log_debug(plugin, fmt, ...) \
    log_write(LOG_LEVEL_DEBUG, plugin, __FILE__, __LINE__, fmt, ##__VA_ARGS__)

#define log_info(plugin, fmt, ...) \
    log_write(LOG_LEVEL_INFO, plugin, __FILE__, __LINE__, fmt, ##__VA_ARGS__)

#define log_warn(plugin, fmt, ...) \
    log_write(LOG_LEVEL_WARN, plugin, __FILE__, __LINE__, fmt, ##__VA_ARGS__)

#define log_error(plugin, fmt, ...) \
    log_write(LOG_LEVEL_ERROR, plugin, __FILE__, __LINE__, fmt, ##__VA_ARGS__)

#define log_fatal(plugin, fmt, ...) \
    log_write(LOG_LEVEL_FATAL, plugin, __FILE__, __LINE__, fmt, ##__VA_ARGS__)

/* ============================================================
   底层日志写入函数（通过 IPC 调用 LOG 插件）
   ============================================================ */
static inline void log_write(log_level_t level, const char *plugin,
                              const char *file, int line,
                              const char *fmt, ...) {
    log_write_req_t req;
    va_list args;
    
    req.level = level;
    strncpy(req.plugin, plugin, sizeof(req.plugin) - 1);
    strncpy(req.file, file, sizeof(req.file) - 1);
    req.line = line;
    
    va_start(args, fmt);
    vsnprintf(req.msg, sizeof(req.msg), fmt, args);
    va_end(args);
    
    // 通过 IPC 发送到 LOG 插件
    ipc_send(LOG_TASK_ID, &req, sizeof(req));
}
```

十一、总结

维度 决策
定位 日志中心，所有日志的统一管理入口
写入方式 异步队列（非阻塞，不拖慢业务）
队列大小 4096 条消息
日志级别 debug < info < warn < error < fatal
输出目标 文件 + 控制台 + 串口（可配置）
日志轮转 按大小轮转（默认 10MB），保留 5 个历史文件
审计日志 单独文件，记录 warn/error/fatal 级别
配置方式 通过 PARAM 系统
依赖 微内核（IPC、任务、内存）+ METAPROTO + PARAM

LOG 是 HWRun OS 中 BUS 加载的第二个功能性插件（PARAM → LOG）。它提供异步、非阻塞的日志写入服务，支持多输出目标、自动轮转和审计日志。所有插件通过 LOG 记录日志，日志级别可动态调整，确保系统运行可观测、可追溯。
